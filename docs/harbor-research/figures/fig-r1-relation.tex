% Worked covering for Theorem 1: N=6, k=1, m=2.
% Three disjoint 2-sets cover all six placements with three of four
% available 2-bit messages. This instance attains the counting floor
% after integer rounding; the counting inequality is only necessary in general.
\begin{figure}[t]
\centering
\begin{tikzpicture}[x=1cm,y=1cm]
  \node[font=\normalsize\bfseries,text=harborblue] at (5.7,6.05)
    {R1 --- a digest message selects a set to open};

  \draw[rounded corners=3pt,draw=harborblue!65,fill=harborblue!7]
    (0.1,1.75) rectangle (5.45,5.62);
  \draw[rounded corners=3pt,draw=seagreen!75,fill=seagreen!7]
    (5.95,1.75) rectangle (11.3,5.62);

  \node[font=\small\bfseries,text=harborblue] at (2.775,5.17)
    {Possible critical locations};
  \node[font=\small] at (2.775,4.55)
    {$N=6$ artifacts; $k=1$ critical};
  \foreach \n/\x in {1/0.85,2/1.62,3/2.39,4/3.16,5/3.93,6/4.70}{
    \fill[harborblue] (\x,3.55) circle (2.4pt);
    \node[font=\small] at (\x,3.15) {\n};
  }
  \node[font=\small] at (2.775,2.35)
    {$\binom{6}{1}=6$ possible placements};

  \node[font=\small\bfseries,text=seagreen!60!black] at (8.625,5.17)

    {Here $B\ge\log_2 3\approx1.585$ bits; three of four two-bit messages suffice.};

\caption{A complete covering in the $N=6$, $k=1$, $m=2$ case.
The three displayed messages name disjoint sets and cover every placement.
The counting inequality is necessary for every codebook; overlap between
sets can prevent a codebook from attaining it in other cases.}
