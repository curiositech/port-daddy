# Expected output of: z3 delta-threshold.z3
#
# Z3 returns the root in algebraic-number form: (root-obj P k) names the
# k-th real root of polynomial P. The polynomial here is the IC cubic
# 2*x^3 + 2*x^2 + 2*x - 1, and k=1 picks the unique real root in (0, 1).
# Numerically delta* ≈ 0.3425080314 (computed via numpy.roots).
#
# Tested with Z3 4.13 and Z3 4.15. Wall-clock < 100 ms.

sat
(
  (define-fun delta () Real
    (root-obj (+ (* 2 (^ x 3)) (* 2 (^ x 2)) (* 2 x) (- 1)) 1))
)
sat
(
  (define-fun delta () Real
    (root-obj (+ (* 2 (^ x 3)) (* 2 (^ x 2)) (* 2 x) (- 1)) 1))
)
unsat

#                 numerical value delta* ≈ 0.3425080314).
#   3rd `unsat` — no SECOND distinct root in [0, 1] (uniqueness).
#
