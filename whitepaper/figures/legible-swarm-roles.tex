\begin{tikzpicture}[
  role/.style={draw=hhink,line width=.75pt,fill=hhsand!26,rounded corners=2pt,

                    font=\tiny,align=center,text=hhink},
]
  \node[role] (operator) {\textbf{Operator}\\\textsc{author}\\\emph{issues and revokes}};
  \node[role,right=11mm of operator] (scope)
    {\textbf{Scoped grant}\\\emph{revocable by operator}};
  \node[role,right=11mm of scope] (daemon)
    {\textbf{Daemon}\\\textsc{sovereign actor}\\\emph{acts legibly}};
  \node[role,right=11mm of daemon] (agents)
    {\textbf{Agents}\\\textsc{the multitude}\\\emph{mutual covenant}};

  \draw[grant] (operator) -- (scope);
  \draw[grant] (scope) -- node[edgelabel] {authorizes} (daemon);
  \draw[act] (daemon) -- (agents);
\end{tikzpicture}
\caption{Authority has one readable route: the operator issues a scoped grant,
which authorizes the daemon to act over the agent multitude.  The operator alone
can revoke the grant.  The agents' covenant is internal to the multitude; it is
not a return arrow from subjects to the sovereign.}
\label{fig:roles}
