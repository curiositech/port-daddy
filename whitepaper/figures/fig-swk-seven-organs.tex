% ─── Six organs in a 3x2 grid ─────────────────────────────────────────────
\node[organ] (resource) at (0,0)
  {{\bfseries Resource}\\ ports, file/region/symbol claims, named locks.\\
   {\footnotesize\textsf{implemented}}};
\node[organ,right=4mm of resource] (identity)
  {{\bfseries Identity}\\ semantic actor names (project\,:\,stack\,:\,context); heartbeat.\\
   {\footnotesize\textsf{implemented} (self-asserted)}};
\node[organ,right=4mm of identity] (comm)
  {{\bfseries Communication}\\ message bus, tuple space, stigmergic markers, two transports.\\
   {\footnotesize\textsf{implemented}}};

\node[organ,below=4mm of resource] (oblig)
  {{\bfseries Obligation \& enforcement}\\ a policy monitor (regiment/detect) + commitments (enforce).\\
   {\footnotesize\textsf{implemented}\,/\,\textsf{partial}}};
\node[organ,below=4mm of identity] (cont)
  {{\bfseries Continuity}\\ sessions, notes, recovery, episodic memory, audit log.\\
   {\footnotesize\textsf{implemented}\,/\,\textsf{partial}}};
\node[organ,below=4mm of comm] (attest)
  {{\bfseries Self-attestation}\\ an invariant registry; boot gate, pre-flight, watchdog.\\
   {\footnotesize\textsf{implemented}}};

% ─── The substrate floor ──────────────────────────────────────────────────
\node[sub,below=6mm of cont] (substrate)
  {{\bfseries\large Substrate organ --- the single-writer transactional floor}\\[2pt]

   a bounded busy-wait, and foreign-key enforcement; a runtime-shim layer keeps the
   deployment and test SQLite bindings in step.\\
   {\footnotesize\textsf{implemented} (durability split I1a/I1b --- see \S\ref{sec:invariants})}};

\caption{The kernel as \emph{seven organs}, each a contract the daemon must hold
for the layers above to be coherent. Six organs are, mechanically, just tables
\emph{with discipline} over a single shared file --- the \textbf{substrate organ},
the single-writer transactional floor. The substrate's storage configuration
\emph{is} the kernel's durability claim; everything else is convenience built on top
of it. Maturity grades (\textsf{implemented} $\cdot$ \textsf{partial} $\cdot$
\textsf{specified} $\cdot$ \textsf{proposed}) are per organ; note that the identity
organ is implemented but \emph{self-asserted}, the gap a cross-machine economy must
eventually close with cryptographic identity.}
\label{fig:swk-seven-organs}
