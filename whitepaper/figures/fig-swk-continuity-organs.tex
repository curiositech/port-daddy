\begin{figure}[H]
\centering
\begin{tikzpicture}[pd figure,x=1.28cm,y=1.0cm]

  \node[pd axis label,rotate=90,anchor=south,text width=3.9cm,align=center]

\caption{The continuity chain as an evidence-support profile. What checkpoint

weakest upstream link. [internal]}
