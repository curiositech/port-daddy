% Figure II.5 — guarded lifecycle with the absent QUEUED state made explicit.
% Visual question: which state transitions exist, and which fairness state does not?
\begin{figure}[H]
\centering
\resizebox{0.90\textwidth}{!}{%
\begin{tikzpicture}[
  font=\small,
  state/.style={draw=hhink,line width=.6pt,fill=hhsand!22,rounded corners=2pt,
    inner sep=8pt,minimum width=31mm,minimum height=12mm,align=center,
    font=\footnotesize\bfseries},
  held/.style={state,draw=hhcobalt,line width=1pt,fill=hhpaper},
  absent/.style={draw=hhgray,dashed,line width=.6pt,fill=none,rounded corners=2pt,
    inner sep=8pt,minimum width=31mm,minimum height=12mm,align=center,
    font=\footnotesize\bfseries,text=hhgray},
  route/.style={-{Stealth[length=2.3mm,width=1.5mm]},draw=hhink,line width=.65pt},
  accentroute/.style={route,draw=hhcobalt,line width=.8pt},
  band/.style={font=\scriptsize\itshape,text=hhgray,align=center},
]
\node[state] (free) at (-4.5,0) {FREE};
\node[held] (held) at (0,0) {HELD by $a$};
\node[state] (expired) at (4.5,0) {EXPIRED};
\draw[accentroute] (free) -- (held);
\node[band] at (-2.25,.75) {atomic acquire};
\draw[route] (held.north) -- ++(0,1.05) -| (free.north);
\node[band] at (-2.25,1.45) {idempotent release};
\draw[route] (held) -- (expired);
\node[band] at (2.25,.75) {TTL elapses};
\draw[route] (expired.south) -- ++(0,-1.05) -| (free.south);
\node[band] at (0,-1.45) {lazy sweep on next read};
\draw[accentroute] (held.south) -- ++(0,-2.2);
\node[absent] (queued) at (0,-3.0) {QUEUED};
\node[band,anchor=west,text=hhcobalt] at (1.85,-3.0)
  {absent: no wait-list or bounded fairness};
\end{tikzpicture}}
\caption{\textbf{Claim lifecycle and the missing fairness state.} Acquire is one
atomic transition, release is idempotent, and expiry is reclaimed lazily. The
dashed \textsc{queued} socket is intentionally unreachable: the current kernel has
no wait-list, so a fast claimant can repeatedly beat a slow one. OP-1 is therefore
visible as missing state, not buried in a captioned loop.}
\label{fig:swk-claim-lifecycle}
\end{figure}
