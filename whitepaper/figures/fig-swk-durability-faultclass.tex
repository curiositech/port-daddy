% Figure II.4 — the same successful commit judged against two fault classes.
% Visual question: which persistence stratum survives each fault?
\begin{figure}[H]
\centering
\resizebox{0.90\textwidth}{!}{%
\begin{tikzpicture}[
  font=\small,
  cell/.style={draw=hhink,line width=.55pt,fill=hhsand!22,rounded corners=2pt,
    inner sep=7pt,minimum width=55mm,minimum height=11mm,align=center},
  exposure/.style={cell,draw=hhcobalt,line width=.95pt,fill=hhpaper},
  band/.style={font=\footnotesize\bfseries,align=center},
  note/.style={font=\footnotesize\itshape,text=hhgray,align=center},
]
\node[band] at (-3.4,2.75) {PROCESS FAILURE};
\node[band] at (3.4,2.75) {OS / POWER FAILURE};
\node[cell] (pdaemon) at (-3.4,1.55) {daemon memory\\{\scriptsize process terminates}};
\node[cell] (pcache) at (-3.4,0) {OS page cache\\{\scriptsize still resident}};
\node[cell] (pdisk) at (-3.4,-1.55) {WAL on storage\\{\scriptsize flush may follow}};
\node[cell] (odaemon) at (3.4,1.55) {daemon memory\\{\scriptsize lost}};
\node[exposure] (ocache) at (3.4,0) {OS page cache\\{\scriptsize unsynced bytes may vanish}};
\node[cell] (odisk) at (3.4,-1.55) {WAL on storage\\{\scriptsize only prior syncs survive}};
\draw[-{Stealth[length=2.2mm]},hhink,line width=.65pt] (pdaemon) -- (pcache);
\draw[-{Stealth[length=2.2mm]},hhink,line width=.65pt] (pcache) -- (pdisk);
\draw[-{Stealth[length=2.2mm]},hhink,line width=.65pt] (odaemon) -- (ocache);
\draw[-{Stealth[length=2.2mm]},hhink,line width=.65pt] (ocache) -- (odisk);
\node[note,anchor=north] at (-3.4,-2.35) {I1a: committed state survives};
\node[note,anchor=north,text=hhcobalt] at (3.4,-2.35) {I1b: latest commits are not guaranteed};
\end{tikzpicture}}
\caption{\textbf{Durability is indexed by fault class.} A process failure leaves
the operating-system cache alive, satisfying I1a. A power or operating-system
failure erases unsynced cache state, so under normal WAL synchronization I1b does
not hold for the latest commits. The highlighted cell is the precise exposure;
crash-consistency must not be sold as power-loss durability.}
\label{fig:swk-durability-faultclass}
\end{figure}
