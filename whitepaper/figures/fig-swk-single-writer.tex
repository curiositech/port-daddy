% Figure 3 — concurrent arrival intervals collapse into one commit order.
\begin{figure}[H]
\centering
\begin{tikzpicture}[pd figure,x=.72cm,y=.72cm]
  % row hairlines and labels
  \foreach \y/\lbl in {7.4/{CLI},6.4/{tool bridge},5.4/{binary IPC},4.4/{worker}} {
    \node[pd row label] at (-.25,\y) {\lbl};
    \draw[pd hairline] (0,\y)--(9.4,\y);
  }
  \node[pd row label,text width=1.7cm,align=right] at (-.25,2.2) {one writer};

  % bands: same neutral fill for every caller, no special tint for the commit
  \filldraw[pd neutral fill] (.5,7.2) rectangle (4.2,7.6);
  \filldraw[pd neutral fill] (1.4,6.2) rectangle (6.0,6.6);
  \filldraw[pd neutral fill] (2.4,5.2) rectangle (7.4,5.6);
  \filldraw[pd neutral fill] (3.6,4.2) rectangle (8.4,4.6);

  % rail
  \draw[pd focus rule,line width=1.3pt] (.5,2.2)--(9.0,2.2);

  % commit marks and order numbers on the rail
  \foreach \x/\n in {4.2/1,6.0/2,7.4/3,8.4/4} {
    \node[pd focus datum] at (\x,2.2) {};
    \node[pd axis label,anchor=north] at (\x,1.82) {\n};
  }

  % replies: one upward arrow from each commit straight back into its
  % caller's row -- the arrow itself is the only line needed between the
  % commit and the band; a separate guide line tracing the same segment
  % downward would just double the ink for no new information.
  \foreach \x/\rowtop in {4.2/7.2,6.0/6.2,7.4/5.2,8.4/4.2} {
    \draw[pd arrow] (\x,2.32)--(\x,\rowtop);
  }
  \node[pd note,anchor=south west,text width=4.4cm,align=left] at (4.55,7.65) {reply after commit};

  % time axis, hairline
  \draw[pd hairline,-{Stealth[length=1.65mm,width=1.05mm]}] (.3,1.0)--(9.05,1.0);
  \node[pd axis label,anchor=west] at (9.1,1.0) {time};

  % model boundary
  \draw[pd boundary] (9.3,2.85) rectangle (14.3,6.55);
  \node[pd note,align=center,text width=2.8cm] at (11.8,4.7)
    {second daemon on the same file: outside this model};
\end{tikzpicture}
\caption{Four concurrent callers and one writer. Each band is a request in
flight; its right edge is where the daemon commits it, and the rail shows
the resulting order 1--4. A reply is sent only after the commit. A second
daemon opening the same file is outside the model, not another lane.
[internal]}
\label{fig:swk-single-writer}
\end{figure}
