% Restyle of fig-state-of-nature to the tikz-diagram-craft v2 standard (v1
% names and raw colours migrated to roles; layout unchanged, ch4.md 4.2).
%
% Reader question: how does consent change concurrent writes into a legible
% order?
% Claim: consent does not shrink who may write; it replaces an unordered
% overlap with one commit spine, each grant explaining itself.
% Evidence: the four write requests A--D and their windows/grants, whitepaper/
% legible-swarm.tex, the state-of-nature section [internal].
% Must distinguish: the top panel's overlap (four valid writers, no order)
% from the bottom panel's spine (same four requests, now ordered by grant).
% Grammar chosen: aligned before/after schedules on one time scale (S1 of the
% atlas row) -- a Gantt pair is the only grammar that keeps both panels on
% the same axis, so the reader compares windows directly.
% Grammar rejected: a membership graph of writers around the artifact would
% show who may write, which nothing here disputes; the claim is about order.
% Five-second test: does consent remove a writer, or just give each a place
% in line?
\begin{figure}[H]
\centering
\pdfigurehue{pdteal}%
\begin{tikzpicture}[pd figure,x=0.68cm,y=0.84cm]
  % ---------------- top panel: unsequenced execution ----------------
  \node[pd title,anchor=west] at (0,4.35) {unsequenced execution};

  % overlap tint, drawn first so the bands sit on top of it
  \fill[pd warn fill] (3,0.41) rectangle (4,3.29);

  \foreach \y/\a/\xa/\xb in {3.05/A/0.5/4,2.25/B/1/5.5,1.45/C/2/6,0.65/D/3/7.5}{
    \node[pd title,anchor=east] at (-0.25,\y) {\a};
    \draw[pd guide] (0,\y) -- (8,\y);
    \fill[pd neutral fill] (\xa,\y-0.24) rectangle (\xb,\y+0.24);
  }
  % re-stroke the overlap edges on top so the bands do not erase them
  \draw[pd warn rule] (3,0.41) -- (3,3.29);
  \draw[pd warn rule] (4,0.41) -- (4,3.29);

  \node[pd warn label,anchor=south] at (3.5,3.52)
    {four valid writes overlap};
  \draw[pd hairline] (3.5,3.44) -- (3.5,3.30);

  \draw[pd rule,-{Stealth[length=1.65mm,width=1.05mm]}] (0,0) -- (8.35,0);
  \foreach \x in {0,2,4,6,8}{
    \draw[pd tick] (\x,0) -- (\x,-0.15);
    \node[pd label,anchor=north] at (\x,-0.20) {\x};
  }
  \node[pd label,anchor=west] at (8.35,0) {time};

  % ---------------- bottom panel: consented serialization ----------------
  \node[pd title,anchor=west] at (0,4.35-6.0) {consented serialization};

  \foreach \y/\a/\xa/\xg in {3.05/A/0.5/2,2.25/B/1/3.5,1.45/C/2/5,0.65/D/3/6.5}{
    \node[pd title,anchor=east] at (-0.25,\y-6.0) {\a};
    \draw[pd guide] (0,\y-6.0) -- (8,\y-6.0);
    \fill[pd neutral fill] (\xa,\y-0.24-6.0) rectangle (\xg,\y+0.24-6.0);
    \draw[pd hairline] (\xg,\y-0.24-6.0) -- (\xg,-6.0);
  }

  \draw[pd rule,-{Stealth[length=1.65mm,width=1.05mm]}] (0,-6.0) -- (8.35,-6.0);
  \foreach \x in {0,2,4,6,8}{
    \draw[pd tick] (\x,-6.0) -- (\x,-6.15);
    \node[pd label,anchor=north] at (\x,-6.20) {\x};
  }
  \node[pd label,anchor=west] at (8.35,-6.0) {time};

  % commit spine: the four grants land on the time axis in order
  \draw[pd focus rule,line width=1.25pt] (2,-6.0) -- (6.5,-6.0);
  \foreach \xg/\lab in {2/g1,3.5/g2,5/g3,6.5/g4}{
    \node[pd focus datum] at (\xg,-6.0) {};
    \node[pd focus label,anchor=north] at (\xg,-6.55) {\lab};
  }
  \node[pd note,anchor=north,align=center,text width=7.6cm] at (4.25,-7.05)
    {each grant records actor $\cdot$ claim $\cdot$ artifact $\cdot$ reason};
\end{tikzpicture}
\caption{Two Gantt panels on one time axis. Top: four write requests A--D
arrive between $t{=}0.5$ and $t{=}3$ with windows that all cover $[3,4]$, so
four valid writers overlap on one artifact with no total order among them.
Bottom: the same four requests under consented serialization, each band now
ending at its own grant g1--g4 ($t{=}2, 3.5, 5, 6.5$) on one commit
spine, and every grant recording actor, claim, artifact, and reason. Consent
changes the schedule, not who may participate [internal].}
\label{fig:state-of-nature}
\end{figure}
