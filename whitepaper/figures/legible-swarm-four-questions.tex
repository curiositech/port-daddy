]
  \node[question] (q1) {\textbf{1. Irreversible act?}};
  \node[obligation,right=16mm of q1] (o1) {\textbf{Force a zoom}\\one keystroke from artifact};
  \node[question,below=7mm of q1] (q2) {\textbf{2. Administrative or m\^etis?}};
  \node[obligation,right=16mm of q2] (o2) {\textbf{Link m\^etis verbatim}\\do not flatten local knowledge};
  \node[question,below=7mm of q2] (q3) {\textbf{3. Can the summary lie?}};
  \node[obligation,right=16mm of q3] (o3) {\textbf{Zoom to the artifact}\\not merely to narration};
  \node[question,below=7mm of q3] (q4) {\textbf{4. Is authority legible?}};
  \node[obligation,right=16mm of q4] (o4) {\textbf{Name the reason}\\each act carries a cause};

  \draw[route] (q4.south) -- (pass.north);
\end{tikzpicture}
\caption{The four design questions are an ordered admission path. Each question binds a

(Prop.~\ref{prop:legible-sovereign}).}
