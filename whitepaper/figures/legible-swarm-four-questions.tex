  result/.style={draw=hhgray!72,line width=.5pt,fill=hhpaper,rounded corners=2pt,

]
  \node[font=\bfseries\small,text=hhink] (title) {RELEASE GATE FOR A READ-SURFACE};
  \node[below=2.5mm of title,font=\scriptsize,text=hhgray] (subtitle)
    {Each check produces an observable condition; all four are required.};

  \node[check,below=7mm of subtitle] (q1)
    {\textbf{1. Irreversible effect?}\\Can this surface authorize a merge, deployment, deletion, or other non-reversible act?};
  \node[result,right=13mm of q1] (r1)
    {\textbf{Required result}\\Artifact-level review is one action away.};

  \node[check,below=6mm of q1] (q2)
    {\textbf{2. Administrative or local judgment?}\\Does the item contain practical context that a summary must not flatten?};
  \node[result,right=13mm of q2] (r2)
    {\textbf{Required result}\\Preserve the relevant context verbatim.};

  \node[check,below=6mm of q2] (q3)
    {\textbf{3. Can the summary misrepresent?}\\Is the displayed claim independently checkable against its source artifact?};
  \node[result,right=13mm of q3] (r3)
    {\textbf{Required result}\\Link the claim directly to its artifact.};

  \node[check,below=6mm of q3] (q4)
    {\textbf{4. Is authority traceable?}\\Does every coordinator action expose a named reason and scope?};
  \node[result,right=13mm of q4] (r4)
    {\textbf{Required result}\\Record a named reason for the action.};

  \draw[route] (q1.south) -- (q2.north);
  \draw[route] (q2.south) -- (q3.north);
  \draw[route] (q3.south) -- (q4.north);
  \draw[route] (q1.east) -- (r1.west);
  \draw[route] (q2.east) -- (r2.west);
  \draw[route] (q3.east) -- (r3.west);
  \draw[route] (q4.east) -- (r4.west);

  \node[terminal,below=8mm of q4] (release)
    {\textbf{Release decision}\\Release only when every required result is present; otherwise return the surface for correction.};
  \draw[route] (q4.south) -- (release.north);
\end{tikzpicture}
\caption{\textbf{A read-surface is released only after four checkable conditions hold.}
The gate requires artifact-level review for irreversible effects, verbatim preservation of local judgment, an independently checkable artifact link for every fallible summary, and a named reason for every authority action. Any missing condition returns the surface for correction. This is the operational form of the one law (Def.~\ref{def:zoom}) and the legible-sovereign rule (Prop.~\ref{prop:legible-sovereign}).}

\end{figure}
