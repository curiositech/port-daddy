% Counting lower bounds for one reader, two separate readers, and their union.
%
% Stable atlas ID: I/fig:readpoverty (fig:readpoverty)
% Reader question: how do the per-reader, summed, and joint counting bounds
%   differ when the critical sets are disjoint?
%
% Analytic source, not measurements or achieved costs:
%   Floor(N,k,m) = log_2 binom(N,k) - log_2 binom(m,k), k <= m < N.
%   The two-reader comparison additionally requires 2k <= m because the joint
%   head uses Floor(N,2k,m); that condition is not part of the one-head floor.
%   For the worked instance k=2,m=8 and integer N in [16,100],
%   F_1 = log_2(N(N-1)/56) is one head per reader,
%   2F_1 is the two reader-specific heads in total, and
%   F_2 = log_2(N(N-1)(N-2)(N-3)/1680) is one joint head for both readers.
%   At N=60: F_1=5.9822, 2F_1=11.9644, F_2=12.7662,
%   F_2/(2F_1)=1.0670. The ratio F_2/F_1=2.1340 is not the two-head penalty.
% Opening policy: each split head may open m=8 items independently; one joint
% head may open m=8 items total. This is not an equal-total-opening benchmark.
% Source: whitepaper/legible-swarm.tex:2531--2557; provenance a7_experiment.py.
% The curves are authored analytic evaluations of the closed forms, not
% empirical observations or simulation traces.

\begin{figure}[H]
\centering
\pdfigurehue{pdteal}%
\begin{tikzpicture}[pd figure,
  declare function={
    perhead(\n) = ln(\n*(\n-1)/56)/ln(2);
    joint(\n)   = ln(\n*(\n-1)*(\n-2)*(\n-3)/1680)/ln(2);
  }]
\begin{axis}[pd axis,
  width=6.8cm,height=5.4cm,
  xmin=16,xmax=100,ymin=0,ymax=17.5,
  xtick={20,40,60,80,100},
  ytick={0,5,10,15},
  xlabel={swarm size $N$},
  ylabel={information floor (bits)},
  axis on top,clip=false]
  % One reserved numeric rail; the labels below are not placed on curves.
  \node[pd label,anchor=south west,align=left,text width=6.6cm]
    at (rel axis cs:0,1.03)
    {$k{=}2$, $m{=}8$. At $N{=}60$: $F_1{=}5.982$/head;
     $2F_1{=}11.964$ total;
     $F_2{=}12.766$ joint; $F_2/(2F_1){=}1.067\times$.};
  % Analytic integer-domain sweep; smooth=false preserves the authored
  % integer evaluations instead of suggesting measured interpolation.
  \addplot[pd rule,line width=.9pt,domain=16:100,samples=85,smooth=false]
    {perhead(x)};
  \addplot[pd focus rule,line width=1.05pt,domain=16:100,samples=85,smooth=false]
    {2*perhead(x)};
  \addplot[pd focus rule,densely dashed,line width=1.05pt,
    domain=16:100,samples=85,smooth=false]
    {joint(x)};

  % The source worked point and its honest architecture comparison.
  \draw[pd guide] (axis cs:60,0) -- (axis cs:60,13.55);
  \node[pd datum] at (axis cs:60,{perhead(60)}) {};
  \node[pd focus datum] at (axis cs:60,{2*perhead(60)}) {};
  \node[pd focus datum] at (axis cs:60,{joint(60)}) {};

  % True outside-plot gutter: each label is reached by an endpoint leader and
  % cannot be crossed by its own curve or mistaken for a point annotation.
  \draw[pd rule] (axis cs:100,{perhead(100)}) -- (axis cs:104,7.75);
  \node[pd label,anchor=west] at (axis cs:104.5,7.75) {one head $F_1$};
  \draw[pd focus rule] (axis cs:100,{2*perhead(100)}) -- (axis cs:104,14.20);
  \node[pd focus label,anchor=west] at (axis cs:104.5,14.20)
    {two heads $2F_1$};
  \draw[pd focus rule,densely dashed] (axis cs:100,{joint(100)}) -- (axis cs:104,16.45);
  \node[pd focus label,anchor=west] at (axis cs:104.5,16.45)
    {joint head $F_2$};
\end{axis}
\end{tikzpicture}
\caption{Separate readers may each open eight items; the joint reader may
open eight total. The comparison changes the total opening allowance.
These counting lower bounds are not achieved implementation costs.}
\label{fig:readpoverty}
\end{figure}
