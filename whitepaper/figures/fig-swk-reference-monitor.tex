\resizebox{0.93\textwidth}{!}{%
\begin{tikzpicture}[
  font=\small,
  nodebox/.style={draw=hhink,line width=.55pt,fill=hhsand!22,rounded corners=2pt,
    inner sep=7pt,minimum width=29mm,minimum height=11mm,align=center},
  gate/.style={nodebox,draw=hhcobalt,line width=1pt,fill=hhpaper},
  route/.style={-{Stealth[length=2.3mm,width=1.5mm]},draw=hhink,line width=.65pt},
  accentroute/.style={route,draw=hhcobalt,line width=.8pt},
  band/.style={font=\footnotesize\bfseries,align=center},
  note/.style={font=\scriptsize\itshape,text=hhgray,align=center},
]
\node[band] at (-3.3,2.2) {PRE-COMMIT: REGIMENT};
\node[band] at (4.2,2.2) {POST-COMMIT: DETECT + COMPENSATE};
\node[nodebox] (request) at (-6.0,.5) {request};
\node[gate] (gate) at (-2.9,.5) {constraint /\\admission gate};
\node[nodebox] (wal) at (.7,.5) {committed\\WAL event};
\node[nodebox] (log) at (4.0,.5) {operation log};
\node[nodebox] (monitor) at (7.0,.5) {policy monitor};
\draw[route] (request) -- (gate);
\draw[accentroute] (gate) -- (wal);
\draw[route] (wal) -- (log);
\draw[route] (log) -- (monitor);
\draw[accentroute] (gate.south) -- ++(0,-1.25);
\node[note,text=hhcobalt,anchor=north] at (-2.9,-.9)
  {invalid transition refused; no mutation};
\draw[route] (monitor.south) -- ++(0,-1.25) -| (wal.south);
\node[note,anchor=north] at (4.0,-.9)
  {bad prefix already exists; append repair or halt};
\draw[hhcobalt,line width=.9pt,dashed] (-.8,-1.65) -- (-.8,2.55);
\node[note,anchor=south,text=hhcobalt] at (-.8,2.58) {commit boundary};
\end{tikzpicture}}
\caption{\textbf{The commit boundary separates prevention from enforcement.}
Constraints and admission checks run before commit and can make a transition
unreachable. The policy monitor is downstream of the durable operation log; it can
detect, halt later effects, and append compensation, but it cannot erase the bad
prefix. The two guarantees are not interchangeable.}
\label{fig:swk-reference-monitor}
\end{figure}
