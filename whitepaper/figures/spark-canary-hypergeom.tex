% Margin sparkline: canary operating characteristic curve Pr(detect | m).
% Domain m in [0, 80] spans; p = (c/n)*(1-beta) = 0.04.
% Fits inside the 1.3in book margin column.
% Reader question: how large a leak does uniform canary planting catch, and
%   at what span count does detection become near-certain?
% Claim: detection probability climbs steadily with leak size m and is
%   already near its 0.96 ceiling by m=80 spans.
% Source: thm:sealed-power operating curve.
\begingroup\makeatletter
\ifdefined\pd@typesetpicturebox
  \let\pgfsys@typesetpicturebox\pd@typesetpicturebox
\fi
\begin{tikzpicture}[pd figure,x=0.80pt,y=70pt]
  % Baseline and ceiling: neutral references, never a drawn edge.
  \draw[pd hairline] (0,0) -- (80,0);
  \draw[pd guide] (0,1) -- (80,1);
  % Operating curve 1 - (1 - 0.04)^m.
  \draw[pd legible,line width=.9pt] plot[domain=0:80,samples=41]
    (\x,{1 - (1 - 0.04)^\x});
  \foreach \m in {10,25,50,80}{
    \node[pd legible datum] at (\m,{1 - (1 - 0.04)^\m}) {};}
  % Labels: end values, the formula, and the worked range -- the formula
  % sits low and left, in the wedge of clear ground below the curve's early,
  % still-low rise, well clear of both the curve and the ceiling guide.
  \node[pd label,anchor=east] at (0,0) {$0$};
  \node[pd label,anchor=south west] at (80,0.96) {$.96$};
  \node[pd direct label,anchor=south west] at (6,0.08) {$\Pr(\text{detect}\mid m)$};
  \node[pd label,anchor=north] at (40,0) {$m: 0\longrightarrow 80$};
\end{tikzpicture}
\makeatother\endgroup
