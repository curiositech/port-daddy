% Reader question: when can Maya and the routing agent share one ranking?
% Claim: a strict reversal, not a difference in units or a tie, prevents
% one order from satisfying both readers at every selection budget.
% Evidence: synthetic ordinal ranks over three proposals named for real
% lib/ artifacts; the complete data is pdRankSamples below. Checked by
% tests/harbor-research/test_legible_swarm_split_ranker.py; no measured scores.
% Must distinguish: readers (left/right) from alternative cases (top/bottom);
% rank 1 is highest on BOTH sides. Only the briefing/attention pair reverses.
% Grammar: paired slopegraphs, identical geometry and direct endpoint labels.
% Reject: before/after, raw fit-versus-loss units, or an invented unified score.
% Five-second test: Maya puts attention.ts first while the router puts
% briefing.ts first; resurrection.ts remains third for both.
% Width: the 4.5-inch Book column, unscaled. All type inherits shared roles.
% Design references: https://charliepark.org/slopegraphs/ (ordinal scales,
% ties, direct labels); https://www.edwardtufte.com/notebook/
% sparkline-theory-and-practice-edward-tufte/ (words and graphics together).
\begin{figure}[H]
\centering
\pdfigurehue{pdteal}%
\begin{tikzpicture}[pd figure,x=1cm,y=1cm]
  \def\lx{3.45}
  \def\rx{7.65}
  \def\rankpitch{0.84}
  \def\panelpitch{3.70}
  % artifact / routing rank / Maya agreement rank / Maya dispute rank /
  % shared line role / shared endpoint role. These ranks are illustrative.
  \def\pdRankSamples{%
    briefing.ts/1/1/2/pd focus rule/pd focus datum,
    attention.ts/2/2/1/pd rule/pd datum,
    resurrection.ts/3/3/3/pd hairline/pd datum}

  \node[pd label,anchor=west,inner sep=0pt] at (0,1.65)
    {Synthetic ordinal ranks; 1 is highest on both sides};

  \foreach \panel/\heading in {
    0/{Agreement: one order serves both},
    1/{Dispute: the top choices conflict}}{
    \begin{scope}[yshift=-\panel*\panelpitch cm]
      \node[pd title,anchor=west,inner sep=0pt] at (0,1.02) {\heading};
      \node[pd label,anchor=east,inner sep=0pt] at (\lx-.22,.46)
        {Routing agent: fit};
      \node[pd label,anchor=west,inner sep=0pt] at (\rx+.22,.46)
        {Maya: regret if ignored};

      \foreach \artifact/\route/\agreement/\dispute/\stroke/\mark in \pdRankSamples {
        \pgfmathtruncatemacro{\readerRank}{ifthenelse(\panel==0,\agreement,\dispute)}
        % Both ends use the same downward ordinal scale; no arrow implies time.
        \draw[\stroke] (\lx,{-(\route-1)*\rankpitch})
          -- (\rx,{-(\readerRank-1)*\rankpitch});
        \node[\mark] at (\lx,{-(\route-1)*\rankpitch}) {};
        \node[\mark] at (\rx,{-(\readerRank-1)*\rankpitch}) {};
        \node[pd label,anchor=east,align=right,text width=3.15cm,inner sep=0pt]
          at (\lx-.22,{-(\route-1)*\rankpitch})
          {\texttt{\artifact}\quad\route};
        \node[pd label,anchor=west,align=left,text width=3.15cm,inner sep=0pt]
          at (\rx+.22,{-(\readerRank-1)*\rankpitch})
          {\readerRank\quad\texttt{\artifact}};
      }
    \end{scope}
  }
\end{tikzpicture}
\caption{One pair crosses when Maya and the router disagree.
Panels compare alternative preferences, not time.
[internal, synthetic ordinal example]}
\label{fig:split-ranker}
\end{figure}
