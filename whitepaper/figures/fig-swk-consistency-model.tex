% Figure II.11 — concurrent calls linearize at one commit rail.
% Visual question: where does each overlapping operation take effect?
\begin{figure}[H]
\centering
\resizebox{0.92\textwidth}{!}{%
\begin{tikzpicture}[
  font=\small,
  lane/.style={draw=hhgray,line width=.55pt},
  op/.style={draw=hhink,line width=.55pt,fill=hhsand!22,rounded corners=2pt,
    inner sep=5pt,minimum width=25mm,minimum height=8mm,font=\scriptsize,align=center},
  outcome/.style={font=\scriptsize\bfseries,align=left},
  band/.style={font=\footnotesize\itshape,text=hhgray,align=center},
]
\node[anchor=east,font=\footnotesize\bfseries] at (-6.3,1.6) {Agent A};
\node[anchor=east,font=\footnotesize\bfseries] at (-6.3,0) {Agent B};
\node[anchor=east,font=\footnotesize\bfseries] at (-6.3,-1.6) {Agent C};
\foreach \y in {1.6,0,-1.6}{\draw[lane] (-6.0,\y) -- (6.2,\y);}
\node[op] (a1) at (-3.9,1.6) {claim $r$};
\node[op] (b1) at (-2.8,0) {claim $r$};
\node[op] (a2) at (1.1,1.6) {release $r$};
\node[op] (c1) at (2.0,-1.6) {claim $r$};
\draw[hhcobalt,line width=1pt] (-.55,-2.4) -- (-.55,2.4);
\node[band,anchor=south,text=hhcobalt] at (-.55,2.45) {single commit rail};
\foreach \y/\lab in {1.6/$t_1$,0/$t_2$,-.8/$t_3$,-1.6/$t_4$}{
  \fill[hhcobalt] (-.55,\y) circle (2.1pt);
  \node[font=\scriptsize,anchor=west,text=hhcobalt] at (-.35,\y) {\lab};
}
\draw[-{Stealth[length=2.1mm]},hhink,line width=.6pt] (a1.east) -- (-.75,1.6);
\draw[-{Stealth[length=2.1mm]},hhink,line width=.6pt] (b1.east) -- (-.75,0);
\draw[-{Stealth[length=2.1mm]},hhink,line width=.6pt] (a2.west) -- (-.35,-.8);
\draw[-{Stealth[length=2.1mm]},hhink,line width=.6pt] (c1.west) -- (-.35,-1.6);
\node[outcome,anchor=west] at (4.1,1.6) {$t_1$: A wins};
\node[outcome,anchor=west] at (4.1,.55) {$t_2$: B rejected};
\node[outcome,anchor=west] at (4.1,-.55) {$t_3$: A releases};
\node[outcome,anchor=west] at (4.1,-1.6) {$t_4$: C wins};
\end{tikzpicture}}
\caption{\textbf{Concurrent operations become one serial history at commit.}
Agent calls may overlap on their own lanes, but every successful or rejected
claim receives one linearization point on the daemon's commit rail. The resulting
order respects real time without a second decision-maker or agreement protocol.}
\label{fig:swk-consistency-model}
\end{figure}
