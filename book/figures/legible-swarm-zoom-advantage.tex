% Reader question: what does the worst-case query certificate guarantee?
% Claim: F/[2k ceil(log2(F/k))+4k] exceeds one iff d=k/F<1/12.
% Evidence: thm:zoom-advantage; exact formula below, separate m intervals.
% The curve is piecewise hyperbolic, not a step function. The threshold is
% not an observed crossover. No measured point is mixed into this bound.
% Grammar: log-density plot with exact dyadic segments and two worked bounds.
\begin{figure}[H]
\centering
\pdfigurehue{pdteal}%
\begin{tikzpicture}[pd figure]
\begin{axis}[pd axis,
  width=9.3cm,height=6.3cm,
  xmode=log,xmin=0.002,xmax=0.4,ymin=0,ymax=23,
  xtick={0.002,0.004,0.015625,0.0833333333,0.4},
  xticklabels={0.002,0.004,1/64,1/12,0.4},
  ytick={1,5,10,15,20},
  xlabel={positive density $d=k/F$ (log scale)},
  ylabel={guaranteed query ratio $F/Q$},
  clip=false,axis on top]
  \addplot[pd guide,domain=0.002:0.4,samples=2] {1};
  % On [2^-m,2^-(m-1)), ceil(log2(1/d)) = m. Draw independent
  % analytic segments; never join the jumps with interpolation.
  \addplot[pd focus rule,domain=0.002:0.00390624,samples=35] {1/(22*x)};
  \addplot[pd focus rule,domain=0.00390625:0.00781249,samples=35] {1/(20*x)};
  \addplot[pd focus rule,domain=0.0078125:0.01562499,samples=35] {1/(18*x)};
  \addplot[pd focus rule,domain=0.015625:0.03124999,samples=35] {1/(16*x)};
  \addplot[pd focus rule,domain=0.03125:0.06249999,samples=35] {1/(14*x)};
  \addplot[pd focus rule,domain=0.0625:0.12499999,samples=35] {1/(12*x)};
  \addplot[pd focus rule,domain=0.125:0.24999999,samples=35] {1/(10*x)};
  \addplot[pd focus rule,domain=0.25:0.4,samples=35] {1/(8*x)};
  \draw[pd guide] (axis cs:0.0833333333,0) -- (axis cs:0.0833333333,1);
  \node[pd focus datum] at (axis cs:0.0833333333,1) {};
  \node[pd label,anchor=south,text width=3.4cm] at (axis cs:0.12,3)
    {certificate equals $1$\\at $d=1/12$};

\end{axis}
\end{tikzpicture}
\caption{A sufficient certificate, not a measured performance curve.
The theorem's ratio is piecewise hyperbolic, with jumps at inverse powers of
two. Below $d=1/12$ this bound guarantees fewer queries than flat inspection;
above it, failure of this certificate does not establish a loss. For example,
$16$ evenly spread positives among $128$ items require $127$ queries.}
\label{fig:zoom-advantage}
\end{figure}
