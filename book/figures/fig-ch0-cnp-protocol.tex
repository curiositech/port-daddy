  \tikzset{

    pd legible badge/.style={pd badge,pd legible fill},

      \pgfmathsetmacro{\xm}{1.35+(\a+\b)*\gap/2}

      \draw[\s,-,pd to badge]  ({1.35+\a*\gap},{\yk+0.12}) -- (m\a\b\k);
