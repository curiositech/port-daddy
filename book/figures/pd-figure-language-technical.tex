% Technical edition overrides (v2 role names): the Seventies manual.  Monoline
% ink, halftone dots and hatching for regions, one warm accent (amber) for the
% figure's subject and one caution (rust) for breach, open arrowheads, square
% corners, grotesk labels.  Concept hues survive only as thin strokes and
% dotted screens, so meaning crosses editions without breaking the monoline.
% Loaded by pd-figure-language.tex before its concept map expands.
\renewcommand{\pdsetfocushue}{\colorlet{pdfocus}{pdamber}}
%
% Typographic law (#10191): an edition changes the FAMILY, in one command, and
% the character of the marks; it never changes the size, and it never restates
% a guide's width without the dash derived from it.
\renewcommand{\pdfiglabelfamily}{\pdgrotesk}
\renewcommand{\pdfigbasesize}{\footnotesize}

% A concept keeps its hue on the stroke and on a dotted screen; breach is
% carried by rust, the edition's one caution ink.
\makeatletter
\renewcommand{\pdconcept}[2]{%
  \def\pd@hue{#2}\def\pd@err{pderror}%
  \ifx\pd@hue\pd@err\def\pd@hue{pdrust}\fi
  \edef\pd@set{\noexpand\tikzset{%
    #1/.style={draw=\pd@hue,text=pdink},
    #1 rule/.style={draw=\pd@hue,line width=1pt},
    #1 arrow/.style={draw=\pd@hue,line width=.8pt,->},
    #1 state/.style={pd state,draw=\pd@hue,line width=.8pt,fill=none,
      pattern={Dots[distance=2.4pt,radius=.35pt]},pattern color=\pd@hue},
    #1 fill/.style={fill=none,pattern={Dots[distance=2.4pt,radius=.35pt]},
      pattern color=\pd@hue,draw=pdink,line width=.4pt},
    #1 datum/.style={circle,fill=\pd@hue,draw=pdink,line width=.4pt,inner sep=2.2pt},
    #1 label/.style={pd label,text=pdink},
  }}\pd@set}
\makeatother
