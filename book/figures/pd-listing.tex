% figures/pd-listing.tex -- code listings a reader can take in at a glance.
%
% A listing is set on a faint sand ground behind a teal rule. Three pieces,
% each optional beyond the listing itself:
%
%   \begin{pdlisting}{KIND}{where it lives}
%   \begin{lstlisting}[caption={...},label={alg:...}]
%   if lease.expired:  (*@\pdco{1}@*)
%   \end{lstlisting}
%   \begin{pdcallouts}
%   \pdnote{1}{\textbf{Expiry before holder.} Why the order matters.}
%   \end{pdcallouts}
%   \end{pdlisting}
%
% THE HEAD names the kind of text (PSEUDOCODE, RUST, PROVERIF, TLA+, OUTPUT)
% and, at the right, the file it is drawn from or the artifact it mirrors.
% The listing's number and caption stay where every other exhibit's are.
%
% THE CALLOUTS replace trailing comments. A comment at the end of a code
% line does not fit the Book's column and wraps under a hook; a numbered
% badge fits, and its note is set below in the text face, where it can say
% in a full sentence which decision the line makes. Use them in pseudocode
% the Book owns. A listing quoted from a model or source file is quoted
% exactly, comments included, and takes no badges.
%
% EXCERPTS. Code from the Book's own kernel is not typed into a chapter. A
% script in the kernel's repository (tools/excerpts.mjs) cuts each marked
% snippet into excerpts/<name>.code, with a one-line excerpts/<name>.tex
% that carries the file, its lines and the language, and copies each
% recorded run to excerpts/<name>.txt. excerpts/manifest.json pins the
% commit and a SHA-256 for every file, and the script's check fails when
% the code, a recording or a file here has changed since.
%
%   \pdexcerpt[caption={...},label={alg:...}]{ch01-claim}
%       sets excerpts/ch01-claim.code under a head that names the language
%       and, at the right, the file and its lines. The numbers beside the
%       code are the file's own. The optional argument takes listings keys.
%       The head prints no commit yet: the data carries the commit of a
%       repository no reader can reach. It gains the public commit when the
%       code is published. A listing's label starts with alg:, the one
%       key-value label the Book's generator gives a chapter prefix.
%   \pdrun[caption={...},label={alg:...}]{lost-update-kernel}
%       sets excerpts/lost-update-kernel.txt, the recorded output of one
%       run, with no line numbers. Its command line is set bold. A listing
%       line holds 66 characters, and the kernel prints its runs at 66.
%   \pdrunlines[caption={...}]{lost-update-no-fence}{1-1,17-32}{1 and 17--32}
%       sets some lines of a run. The head names them, so the reader can see
%       that lines were left out.
%   \pdsessionfile[caption]{lock-demo}
%       sets excerpts/lock-demo.txt in the frame \begin{pdsession} draws. A
%       transcript line holds 54 characters.
%   \pdlistingroom{17}
%       goes before a listing of 17 lines that should not be broken across
%       pages. If the whole listing does not fit on what is left of the
%       page, the page ends here and the listing opens the next one.
%
% \pdco{n} is escaped into a listing with (*@ ... @*), which the shared
% listing style enables. figures/pd-pedagogy.tex loads this file, so every
% chapter has the macros; the colours are the palette's and are looked up
% when a listing is set.
\edef\pdlistingrestoreat{\catcode`\noexpand\@=\the\catcode`\@\relax}
\makeatletter
\providecommand{\pdco}[1]{%
  \tikz[baseline=(pdco.base)]\node[circle,fill=hhteal,inner sep=0.6pt,
    minimum size=8.6pt,font=\sffamily\bfseries\fontsize{6.2}{6.2}\selectfont,
    text=white](pdco){#1};}
\newlength{\pd@listingrule}\setlength{\pd@listingrule}{2.2pt}
% One band of the listing's ground, the rule at its left: #1 is its content,
% set in a \vtop of the remaining width.
\newcommand{\pd@listingband}[1]{%
  \par\nointerlineskip\noindent\begingroup\setlength{\fboxsep}{0pt}%
  {\color{hhteal}\vrule width\pd@listingrule}%
  \colorbox{hhsand!16}{\begin{minipage}[t]{\dimexpr\linewidth-\pd@listingrule\relax}#1\end{minipage}}%
  \endgroup\par\nointerlineskip}
\newif\ifpd@listingclosed
% The listings package ends a displayed listing with \penalty-50, which
% invites a page break after the last line. Every page of the Book is ragged
% at the bottom, so TeX takes that invitation whenever it is the best one
% left on a page, and the closing band then opens the next page alone. The
% penalty has gone to the page by the time this file's code runs and cannot
% be taken back, so the package's macro is changed to read the penalty from
% a name. Inside pdlisting the name forbids the break. Anywhere else it is
% the package's own -50. A listing may still part from its callouts:
% pdcallouts puts the -50 back.
\newcommand{\pd@listingendpenalty}{-50}
\patchcmd{\lst@DeInit}{\par\penalty-50\vspace\lst@belowskip}%
  {\par\penalty\pd@listingendpenalty\relax\vspace\lst@belowskip}{}%
  {\PackageError{pd-listing}{The end of a listing could not be patched}%
    {The listings package no longer ends a display with \string\penalty-50.
     Read \string\lst@DeInit and change the patch in figures/pd-listing.tex.}}
% Room for a whole listing of #1 lines. Each line takes 10.55pt. The head
% and the closing band take 41pt between them, measured from the end of the
% paragraph above, and 44pt leaves a little over. \baselineskip is not a
% safe unit for this: between paragraphs it is 12.6pt, though the body's
% lines are set 14.56pt apart.
\newcommand{\pdlistingroom}[1]{\Needspace{\dimexpr44pt+10.55pt*#1\relax}}
\newsavebox{\pd@listingfoot}
\newenvironment{pdlisting}[2]{%
  % Keep the head with its listing: a captioned listing asks for room for
  % its margin caption and may break the page after the head otherwise.
  \par\ifinner\else\Needspace{9\baselineskip}\fi
  \par\addvspace{9pt}\begingroup\global\pd@listingclosedfalse
  \def\pd@listingendpenalty{\@M}%
  \pd@listingband{\vspace{4pt}\hspace{6pt}%
    {\sffamily\bfseries\fontsize{7.2}{9}\selectfont\textcolor{hhteal}{\MakeUppercase{#1}}}\hfill
    {\ttfamily\fontsize{7}{9}\selectfont\textcolor{hhgray}{#2}}\hspace{6pt}\par\vspace{1pt}%
    \hspace{6pt}{\color{hhgray!40}\rule{\dimexpr\linewidth-12pt\relax}{0.3pt}}\par\vspace{2.5pt}}%
  % The listing style puts 7pt above a listing and 5pt below it. Inside this
  % environment the bands abut the listing, so both are taken back: above,
  % behind a zero rule so that listings' \addvspace cannot cancel it.
  \nobreak\vskip-7pt\nobreak\nointerlineskip
  % A margin caption can leave a point or two between the head and the first
  % line. Paint the ground and the rule 4pt down from the head first (the
  % head ends 7pt below this point, where the listing skip is restored); the
  % listing is set over them.
  \hbox{\smash{\lower11pt\hbox{{\color{hhteal}\vrule width\pd@listingrule height4pt}%
    {\color{hhsand!16}\vrule width\dimexpr\linewidth-\pd@listingrule\relax height4pt}}}}%
  \nobreak\hrule height0pt\nobreak\nointerlineskip}{%
  \par\ifpd@listingclosed\else
    % A displayed listing ends with a penalty and its 5pt belowskip, which
    % \vspace follows with a zero skip. Inside this environment the penalty
    % forbids a break (\pd@listingendpenalty above), so nothing can part the
    % last line from the closing band. Take the skips back if they are still
    % here to take. If the page has them already, cancel the 5pt instead.
    \ifdim\lastskip=\z@\unskip\fi
    \@tempdima\lastskip
    \ifdim\@tempdima>\z@
      \unskip\ifnum\lastpenalty<\z@\unpenalty\fi\nobreak
    \else
      \nobreak\vskip-5pt
    \fi
  \fi
  % The closing band is set in a box of no height that hangs below the last
  % line, and the room it takes follows as a kern. A page may end at the
  % kern, and TeX then drops the kern. So the band needs no room of its own
  % at the foot of a page, and a listing that fills a page to its last line
  % is not made to send that line over with its band. The skip back by the
  % last line's depth is so that the page, ended there, is no taller than
  % one ended at the last line. The band is set lower by the same amount.
  \@tempdimb\prevdepth\ifdim\@tempdimb<\z@\@tempdimb\z@\fi
  \setbox\pd@listingfoot\vbox{\pd@listingband{\vspace{2pt}\mbox{}}}%
  \@tempdima\dimexpr\ht\pd@listingfoot+\dp\pd@listingfoot\relax
  \nobreak\vskip\parskip\vskip-\@tempdimb\nointerlineskip
  \vbox to\z@{\kern\@tempdimb\box\pd@listingfoot\vss}%
  \kern\dimexpr\@tempdimb+\@tempdima\relax\nointerlineskip
  \endgroup\par\addvspace{9pt}}
\newsavebox{\pd@calloutbox}
\newenvironment{pdcallouts}{%
  \par\nobreak\vskip-5pt\penalty-50\global\pd@listingclosedtrue
  \begin{lrbox}{\pd@calloutbox}\begin{minipage}[t]{\dimexpr\linewidth-\pd@listingrule\relax}%
    \vspace{3pt}%
    \hspace{6pt}{\color{hhgray!40}\rule{\dimexpr\linewidth-12pt\relax}{0.3pt}}\par\vspace{3pt}%
    \leftskip=6pt\rightskip=6pt plus 2em\relax
    \sffamily\fontsize{8.2}{10.6}\selectfont\setlength{\parindent}{0pt}}{%
  \par\end{minipage}\end{lrbox}%
  \par\nobreak\nointerlineskip\noindent\begingroup\setlength{\fboxsep}{0pt}%
  {\color{hhteal}\vrule width\pd@listingrule}\colorbox{hhsand!16}{\usebox{\pd@calloutbox}}%
  \endgroup\par\nointerlineskip}

% The gutter is a little wider than a pseudocode listing's, because a file's
% own line numbers run to three digits. The code column then holds 66
% characters with half a character to spare, and 66 is the width
% tools/excerpts.mjs enforces. Widen the gutter further and a 66-character
% line wraps.
\lstdefinestyle{pdexcerpt}{style=pdcode,language={},tabsize=4,
  framexleftmargin=2.15em,xleftmargin=\dimexpr2.15em+5.2pt\relax}
\newcommand{\pdexcerptdata}[6]{%
  \def\pd@excerpt@path{#2}\def\pd@excerpt@first{#3}\def\pd@excerpt@last{#4}%
  \def\pd@excerpt@commit{#5}\def\pd@excerpt@language{#6}}
% The data file's name is built in a macro first. The figure atlas scanner
% follows every literal \input{...} it finds, and skips one whose path a
% macro supplies.
\newcommand{\pdexcerpt}[2][]{%
  \def\pd@excerpt@file{excerpts/#2}%
  \input{\pd@excerpt@file.tex}%
  \begin{pdlisting}{\pd@excerpt@language}{\expandafter\detokenize\expandafter{\pd@excerpt@path}, lines \pd@excerpt@first\textendash\pd@excerpt@last}%
  \edef\pd@excerpt@options{style=pdexcerpt,language=\pd@excerpt@language,firstnumber=\pd@excerpt@first}%
  \expandafter\lstinputlisting\expandafter[\pd@excerpt@options,#1]{excerpts/#2.code}%
  \end{pdlisting}}
\lstdefinestyle{pdrun}{style=pdoutput,moredelim=[l][\fontseries{b}\selectfont]{\$\ }}
\newcommand{\pdrun}[2][]{%
  \begin{pdlisting}{Output}{sessions/#2.txt}%
  \lstinputlisting[style=pdrun,#1]{excerpts/#2.txt}%
  \end{pdlisting}}
% The Book's kernel, by name. The box keeps the name whole: set in the mono
% face it could otherwise be broken at a line end, and "har-bord" is not a
% command anyone can type.
\providecommand{\pdkernelname}{\mbox{\texttt{harbord}}}
% Some lines of a recorded run. #3 is the listings line range and #4 says the
% same in words for the head, so a reader can see that lines were left out.
\newcommand{\pdrunlines}[4][]{%
  \begin{pdlisting}{Output}{sessions/#2.txt, lines #4}%
  \lstinputlisting[style=pdrun,linerange={#3},#1]{excerpts/#2.txt}%
  \end{pdlisting}}
% The frame is pdsession's own begin and end code, around a file.
\newcommand{\pdsessionfile}[2][]{%
  \endgraf\begingroup
  \let\pd@sessionfamily\pdblockfamily
  \def\pdblockfamily##1{\pd@sessionfamily{##1}\def\pdblock@color{pdteal}}%
  \pdblockbefore{Terminal transcript}%
  \noindent\raisebox{-2pt}{\includegraphics[width=12pt,height=12pt]{figures/lucide/terminal.pdf}}%
  \endgraf\nobreak
  \pd@measurewidefield
  \lstinputlisting[style=pdsession,upquote=true,linewidth=\pd@sessionwidth,xleftmargin=0pt,
    basicstyle=\ttfamily\ifdefined\addfontfeatures\addfontfeatures{Mapping=}\fi\fontsize{8.6}{10.6}\selectfont,
    literate={-}{{\char45}}1,caption={#1}]{excerpts/#2.txt}%
  \pdblockafter{Terminal transcript}%
  \endgroup}
