% Reader question: who is to blame when two people's directions clash, and
% how does the deletion filter decide? Claim: removing either person's
% direction removes the conflict, so neither can be dropped and the blame set
% is {Alice, Bob}. Evidence: the many-authors example -- Alice's obligation to
% migrate db:checkout in minutes [0, 120] and Bob's release freeze on the same
% scope for minutes [0, 4320]; they overlap on [0, 120]. The filter reruns the
% check once without each principal. Must distinguish: the conflict window
% (the overlap) from either interval alone; a principal who is needed for the
% conflict from one who is not. Grammar: interval bars on one minutes axis,
% three rows (both, without Alice, without Bob), axis broken after 180 so the
% three-day freeze fits; rejected: a Venn diagram (an overlap of sets, not of
% time windows) and a table of checker calls (hides that the clash is an
% overlap in time). Five-second test: "together they overlap; take either
% one away and nothing is left to clash."
\begin{figure}[H]
\centering
\pdfigurehue{pdteal}%
\begin{tikzpicture}[pd figure,x=1cm,y=1cm,
  row/.style={pd title,anchor=west},
  says/.style={pd label,anchor=north west,align=left,text width=7.6cm}]
  % Minutes to cm: 0 at x=2.55, 1 min = 6/180 cm up to 180 (x=8.55); the axis
  % breaks over 8.55--8.95 and 4320 sits at x=10.85.
  % Alice's obligation (solid, chapter colour) and Bob's freeze (hatched) are
  % drawn by these macros so all three rows use one geometry.
  \def\alicebar#1{\draw[pd focus fill] (2.55,#1-.15) rectangle (6.55,#1+.15);}
  % The hatching is drawn, not a pattern fill: a PDF pattern tile lands at the
  % page origin and fails figcheck T5.
  \def\hatch#1#2#3{% x0, x1, y centre
    \begin{scope}
      \clip (#1,#3-.15) rectangle (#2,#3+.15);
      \foreach \k in {0,...,60} {
        \pgfmathsetmacro{\xx}{#1-.30+.13*\k}
        \ifdim\xx pt<#2pt \draw[pd hairline] (\xx,#3-.15) -- ++(.30,.30);\fi}
    \end{scope}
    \draw[pd hairline] (#1,#3-.15) rectangle (#2,#3+.15);}
  \def\bobbar#1{\hatch{2.55}{8.55}{#1}\hatch{8.95}{10.85}{#1}}
  % ---- both directions: the windows overlap on minutes 0--120 --------------
  \node[row] at (0,.53) {Both directions};
  \alicebar{.76}
  \bobbar{.30}
  \draw[pd breach rule] (2.47,.06) rectangle (6.63,1.00);
  \node[pd label,anchor=south west] at (2.55,1.04) {Alice:\ must migrate the database, minutes 0--120};
  \node[pd label,anchor=north west] at (2.55,.02) {Bob (owns the database):\ freeze, minutes 0--4320};
  \node[pd breach label,anchor=west] at (6.80,.76) {conflict on minutes 0--120};
  % ---- the deletion filter: rerun the check without each person ------------
  \node[row] at (0,-1.10) {Without Alice};
  \bobbar{-1.10}
  \node[says] at (2.55,-1.33) {no conflict, so Alice stays in the blame set};
  \node[row] at (0,-2.30) {Without Bob};
  \alicebar{-2.30}
  \node[says] at (2.55,-2.53) {no conflict, so Bob stays in the blame set};
  % ---- minutes axis, broken after 180 ---------------------------------------
  \draw[pd rule] (2.55,-3.35) -- (8.55,-3.35);
  \draw[pd rule] (8.95,-3.35) -- (10.85,-3.35);
  \foreach \x in {8.55,8.95} \draw[pd rule] (\x-.09,-3.50) -- (\x+.09,-3.20);
  \foreach \x/\m in {2.55/0,4.55/60,6.55/120,8.55/180,10.85/4320} {
    \draw[pd tick] (\x,-3.35) -- (\x,-3.45);
    \node[pd label,anchor=north] at (\x,-3.50) {$\m$};
  }
  \node[pd label,anchor=north] at (6.70,-3.95) {minutes from now (axis broken after 180)};
\end{tikzpicture}
\caption{Remove either person's direction and the clash disappears, so the
blame set is $\{\text{Alice},\text{Bob}\}$. Bob owns the database, so his
freeze wins. [constructed example]}
\label{fig:audit-blame}
\end{figure}
