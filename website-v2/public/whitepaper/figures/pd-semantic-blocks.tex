\newcommand{\pdblock@frame}{%
  \draw[draw=\pdblock@color,line width=.5pt]
    (frame.south west) rectangle (frame.north east);
  \ifdefstring{\pdblock@color}{pdblockProof}{%
    \fill[\pdblock@color] (frame.north west) rectangle
      ([yshift=-5pt]frame.north east);
    \fill[\pdblock@color] (frame.south east) rectangle
      ([xshift=-3pt,yshift=20pt]frame.south east);
  }{\ifdefstring{\pdblock@color}{pdblockChecked}{%
    \fill[\pdblock@color] (frame.north west) rectangle
      ([xshift=2pt]frame.south west);

\newcommand{\pdblockbefore}[1]{%
  % Allow a wrapped standalone head, frame padding and two opening body lines.
  \par\if@nobreak\else\Needspace{7\baselineskip}\fi
  \begingroup
  \edef\pdblock@expanded{#1}\expandafter\pdblockfamily\expandafter{\pdblock@expanded}%
  \pdblock@inlinebegin
  \begin{tcolorbox}[enhanced,breakable,sharp corners,frame hidden,
    colback=\pdblock@color!6!white,boxrule=.5pt,boxsep=0pt,
    left=11pt,right=11pt,top=10pt,bottom=10pt,
    before skip=9pt plus 2pt,after skip=9pt plus 2pt,
    pad at break*=10pt,lines before break=6,
    overlay unbroken={\pdblock@frame},overlay first={\pdblock@frame},
    overlay middle={\pdblock@frame},overlay last={\pdblock@frame}]
    \clubpenalty=9000 \widowpenalty=9000
    \displaywidowpenalty=9000 \brokenpenalty=9000
}
\newcommand{\pdblockafter}[1]{\par\end{tcolorbox}\endgroup}
