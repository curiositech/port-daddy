% Every split segment retains a complete perimeter cue.
% When unbroken, the perimeter is solid.
% When broken across pages:
% - first segment has a serrated/zigzag bottom edge + continuation arrow;
% - middle segment has zigzag top and bottom edges + continued/continuation arrows;
% - last segment has a zigzag top edge + continued arrow.
\newcommand{\pdblock@frame@unbroken}{%
  \draw[draw=\pdblock@color,line width=.5pt]
    (frame.south west) rectangle (frame.north east);
  \pdblock@accents@top
  \pdblock@accents@bottom
  \pdblock@accents@side
}
\newcommand{\pdblock@frame@first}{%
  \draw[draw=\pdblock@color,line width=.5pt]
    (frame.south west) -- (frame.north west) -- (frame.north east) -- (frame.south east);
  \draw[draw=\pdblock@color,line width=.7pt,decorate,decoration={zigzag,segment length=8pt,amplitude=1.8pt}]
    (frame.south west) -- (frame.south east);
  \node[anchor=south east, font=\pdgrotesk\fontsize{7.5}{9}\selectfont\itshape, text=\pdblock@color, yshift=-4.5pt, xshift=-1pt]
    at (frame.south east) {continues on next page $\,\blacktriangleright$};
  \pdblock@accents@top
  \pdblock@accents@side
}
\newcommand{\pdblock@frame@middle}{%
  \draw[draw=\pdblock@color,line width=.5pt]
    (frame.south west) -- (frame.north west);
  \draw[draw=\pdblock@color,line width=.5pt]
    (frame.south east) -- (frame.north east);
  \draw[draw=\pdblock@color,line width=.7pt,decorate,decoration={zigzag,segment length=8pt,amplitude=1.8pt}]
    (frame.north west) -- (frame.north east);
  \draw[draw=\pdblock@color,line width=.7pt,decorate,decoration={zigzag,segment length=8pt,amplitude=1.8pt}]
    (frame.south west) -- (frame.south east);
  \node[anchor=north west, font=\pdgrotesk\fontsize{7.5}{9}\selectfont\itshape, text=\pdblock@color, yshift=4.5pt, xshift=1pt]
    at (frame.north west) {$\blacktriangleright\,$ continued from previous page};
  \node[anchor=south east, font=\pdgrotesk\fontsize{7.5}{9}\selectfont\itshape, text=\pdblock@color, yshift=-4.5pt, xshift=-1pt]
    at (frame.south east) {continues on next page $\,\blacktriangleright$};
  \pdblock@accents@side
}
\newcommand{\pdblock@frame@last}{%
  \draw[draw=\pdblock@color,line width=.5pt]
    (frame.north west) -- (frame.south west) -- (frame.south east) -- (frame.north east);
  \draw[draw=\pdblock@color,line width=.7pt,decorate,decoration={zigzag,segment length=8pt,amplitude=1.8pt}]
    (frame.north west) -- (frame.north east);
  \node[anchor=north west, font=\pdgrotesk\fontsize{7.5}{9}\selectfont\itshape, text=\pdblock@color, yshift=4.5pt, xshift=1pt]
    at (frame.north west) {$\blacktriangleright\,$ continued from previous page};
  \pdblock@accents@bottom
  \pdblock@accents@side
}

\newcommand{\pdblockbefore}[1]{%
  % Allow a wrapped standalone head, frame padding and two opening body lines.
  \par\if@nobreak\else\Needspace{7\baselineskip}\fi
  \begingroup
  \edef\pdblock@expanded{#1}\expandafter\pdblockfamily\expandafter{\pdblock@expanded}%
  \pdblock@inlinebegin
  \begin{tcolorbox}[enhanced,breakable,sharp corners,frame hidden,
    colback=\pdblock@color!6!white,boxrule=.5pt,boxsep=0pt,
    left=11pt,right=11pt,top=10pt,bottom=10pt,
    before skip=9pt plus 2pt,after skip=9pt plus 2pt,
    pad at break*=10pt,lines before break=6,
    overlay unbroken={\pdblock@frame@unbroken},
    overlay first={\pdblock@frame@first},
    overlay middle={\pdblock@frame@middle},
    overlay last={\pdblock@frame@last}]
    \clubpenalty=9000 \widowpenalty=9000
    \displaywidowpenalty=9000 \brokenpenalty=9000
}
\newcommand{\pdblockafter}[1]{\par\end{tcolorbox}\endgroup}
