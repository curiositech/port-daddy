% Figure IV.15 --- reviewer count traces one purchasable risk-cost frontier.
%
% Reader question: does adding reviewers actually buy less risk, or just more
% invoices?
% Claim: each independently bonded reviewer buys another multiplicative
% reduction in residual risk, at linear cost -- but only if the reviewers
% are independent.
% Evidence: $P_k=(1-d)^k$ at $d=.6$ (k=1..5: 40%, 16%, 6.4%, 2.6%, 1.0%
% residual risk); the correlated-reviewer counterfactual (effective k~1);
% Becker deterrence $\rho dB\geq G$.
% Must distinguish: the geometric risk-reduction frontier (gold, real
% independent reviewers) from the amber counterfactual (five reviewers that
% behave as one, because they share a blind spot).
% Grammar chosen: paired residual-risk/cumulative-cost trend with the worked
% points marked (atlas first choice; SKILL.md's "quantity against a
% parameter" case), all drawn at full size -- the v1 defect was
% resizebox shrinking 8pt labels under the 7pt floor, not the layout.
% Five-second test: a reader can read off that five independent reviewers
% reach about 1% residual risk, and that five correlated ones do not.
\begin{figure}[H]
\centering
\pdfigurehue{pdgold}%
\begin{tikzpicture}[pd figure,x=1cm,y=1cm]
  \node[pd title,anchor=west,text width=10.6cm] at (-0.30,3.55)
    {each independent reviewer buys another multiplicative reduction in
     residual risk};
  \node[pd focus label,anchor=west] at (-0.30,3.10) {$P_k=(1-d)^k$};
  \draw[pd rule,->] (-0.30,-1.55) -- (8.30,-1.55);
  \draw[pd rule,->] (-0.30,-1.55) -- (-0.30,2.85);
  \node[pd label,anchor=north west,text width=3.2cm] at (5.30,-1.70)
    {cumulative cost $k(b+c_B)$};
  \node[pd label,rotate=90,anchor=south,text width=3.2cm,align=center]
    at (-0.70,.65) {residual defect risk (log scale)};
  \draw[pd focus rule,line width=1.45pt]
    (0.80,2.15) -- (2.40,1.50) -- (4.00,.85) -- (5.60,.20) -- (7.20,-.45);
  \foreach \x/\y/\k/\p in {0.80/2.15/1/40\%,2.40/1.50/2/16\%,4.00/.85/3/6.4\%,
                            5.60/.20/4/2.6\%,7.20/-.45/5/1.0\%}{
    \node[pd focus datum,inner sep=2.4pt] at (\x,\y) {};
    \ifnum\k=2
      \node[pd focus label,anchor=west] at ({\x+.20},{\y+.25}) {$k=\k$};
    \else
      \node[pd focus label,anchor=south] at (\x,{\y+.26}) {$k=\k$};
    \fi
    \node[pd label,anchor=north] at (\x,{\y-.26}) {\p};}
  \draw[pd warn rule,line width=1.25pt] (0.80,2.15) -- (7.20,2.15);
  \node[pd warn datum,inner sep=2.6pt] at (7.20,2.15) {};
  \node[pd label,anchor=west,align=left,text width=2.3cm] at (7.30,2.15)
    {five correlated reviewers: effective $k\approx1$};
  \node[pd note,anchor=north west,text width=9.6cm] at (-0.30,-2.85)
    {frontier is valid only with independent blind spots and deterrence
     $\rho dB\geq G$};
\end{tikzpicture}
\caption{\textbf{Purchased assurance is one risk--cost trajectory.} Each gold
point adds one independently bonded reviewer: cumulative cost rises linearly
while residual defect risk falls geometrically. At $d=.6$, three reviewers
leave $6.4\%$ residual risk and five leave about $1\%$. The dashed amber
counterfactual charges for five correlated reviewers but delivers roughly one
reviewer's protection. Independence and Becker deterrence $\rho dB\geq G$
are therefore conditions of the frontier, not footnotes to headcount. [internal]}
\label{fig:he-assurance}
\end{figure}
