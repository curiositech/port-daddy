% Figure III.13 — a later revocation marks one provenance branch invalid.
% Visual question: what changes when an append-only outcome is revoked?
\begin{figure}[H]
\centering
\resizebox{0.91\textwidth}{!}{%
\begin{tikzpicture}[font=\small,node distance=18mm]
  \node[stp box,text width=28mm] (root) {principal\\outcome ledger};
  \node[stp box,text width=26mm,right=of root,yshift=11mm] (o1) {outcome $o_1$\\{\scriptsize valid}};
  \node[stp accent box,text width=26mm,right=of root,yshift=-11mm] (o3) {outcome $o_3$\\{\scriptsize fraudulent}};
  \node[stp box,text width=27mm,right=of o3,yshift=9mm] (score) {derived score\\$s(o_3)$};
  \node[stp box,text width=27mm,right=of o3,yshift=-9mm] (hire) {dependent hire\\decision};
  \node[stp accent box,text width=30mm,right=28mm of score,yshift=-9mm] (t)
    {appended tombstone\\$\bar{o}_3$};
  \draw[stp arrow] (root) -- (o1);
  \draw[stp accent arrow] (root) -- (o3);
  \draw[stp arrow] (o3) -- (score);
  \draw[stp arrow] (o3) -- (hire);
  \draw[stp accent arrow] (t.west) -- (o3.east);
  \draw[hhcobalt,line width=1.2pt] ([xshift=-4mm,yshift=4mm]o3.center) -- ([xshift=4mm,yshift=-4mm]o3.center);
  \draw[hhcobalt,line width=1.2pt] ([xshift=-4mm,yshift=-4mm]o3.center) -- ([xshift=4mm,yshift=4mm]o3.center);
  \node[stp label,below=8mm of t] {append repair; preserve history};
  \node[stp label,above=7mm of score] {descendants recomputed};
\end{tikzpicture}}
\caption{\textbf{A tombstone invalidates a provenance branch without rewriting
history.} The fraudulent outcome remains in the append-only ledger, but the later
tombstone marks it and every derived score or decision for recomputation. The
protocol bounds the window during which descendants can rely on the poisoned
entry; it does not pretend the window was zero.}
\label{fig:tombstone}
\end{figure}
