% Semantic contract: three evidence layers separated by explicit assumption gaps.
\begin{figure}[H]
\centering
\begin{tikzpicture}[
  font=\small,
  layer/.style={draw=hhink,line width=.65pt,fill=hhsand!22,rounded corners=2pt,
    minimum width=38mm,minimum height=27mm,text width=34mm,
    inner sep=6pt,align=center},
  flow/.style={-{Stealth[length=2.2mm]},draw=hhink,line width=.65pt}
]
  \node[layer] (spec) at (-4.7,0) {\textbf{Protocol model}\\[2pt]
    {\footnotesize ProVerif}\\{\scriptsize proves secrecy, correspondence, attenuation}};
  \node[layer] (source) at (0,0) {\textbf{Rust source}\\[2pt]
    {\footnotesize Kani + tests}\\{\scriptsize checks subset logic, bounds, non-panic}};
  \node[layer,draw=hhteal,line width=1pt] (runtime) at (4.7,0)
    {\textbf{Running binary}\\[2pt]{\footnotesize conformance suite}\\
     {\scriptsize checks selected attacks and the FFI boundary}};

  \draw[flow] (spec.east) -- (source.west);
  \draw[flow] (source.east) -- (runtime.west);
  \draw[hhamber,line width=.75pt,dashed] (-2.35,-1.42) -- (-2.35,1.42);
  \draw[hhamber,line width=.75pt,dashed] (2.35,-1.42) -- (2.35,1.42);
  \node[font=\scriptsize\bfseries,text=hhamber,fill=hhpaper,inner xsep=3pt]
    at (-2.35,1.62) {specification gap};
  \node[font=\scriptsize\bfseries,text=hhamber,fill=hhpaper,inner xsep=3pt]
    at (2.35,1.62) {deployment gap};
  \draw[-{Stealth[length=2.2mm]},hhteal,line width=.8pt]
    (runtime.south) -- ++(0,-9mm) -| (spec.south);
  \node[font=\footnotesize\itshape,text=hhteal,below=9mm of source]
    {runtime evidence narrows both gaps; it does not erase them};
\end{tikzpicture}
\caption{Assurance crosses two explicit gaps: a verified protocol model is not
automatically the Rust implementation, and verified source is not automatically the
deployed binary. Runtime evidence flows back across the same boundaries without
turning either assumption into a theorem.}
\label{fig:anchor-verification-stack}
\end{figure}
