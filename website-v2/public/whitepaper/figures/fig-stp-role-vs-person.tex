% Figure III.2 — roles change while one provenance rail persists.
% Visual question: what stays the same as role instances change?
\begin{figure}[H]
\centering
\resizebox{0.93\textwidth}{!}{%
\begin{tikzpicture}[font=\small,node distance=21mm]
  \node[stp heading] (roles) {ROLE INSTANCES};
  \node[stp box,text width=27mm,below=7mm of roles] (r1)
    {navigator\\{\scriptsize obligation set A}};
  \node[stp box,text width=27mm,right=of r1] (r2)
    {cartographer\\{\scriptsize obligation set B}};
  \node[stp box,text width=27mm,right=of r2] (r3)
    {reviewer\\{\scriptsize obligation set C}};
  \node[stp box,text width=27mm,right=of r3] (r4)
    {lookout\\{\scriptsize obligation set D}};

  \draw[hhcobalt,line width=1.2pt] ([yshift=-15mm]r1.south) -- ([yshift=-15mm]r4.south);
  \foreach \r/\o in {r1/pass,r2/pass,r3/fail,r4/pass}{
    \draw[hhcobalt,line width=.9pt] (\r.south) -- ([yshift=-15mm]\r.south);
    \node[font=\scriptsize,anchor=north] at ([yshift=-17mm]\r.south) {\o};
  }
  \node[stp accent box,text width=102mm,anchor=north] (identity)
    at ([yshift=-27mm]$(r1.south)!0.5!(r4.south)$)
    {\textbf{one bound identity} --- the witnessed history follows this rail,
     not any role or process};
\end{tikzpicture}}
\caption{\textbf{A person is the continuous provenance rail beneath changing
roles.} A role is fillable and history-free; each process may occupy a different
role instance. Outcomes accumulate against one bound identity, so reputation
survives role changes and respawns without being attached to an org-chart slot.}
\label{fig:role-person}
\end{figure}
