\begin{tikzpicture}[

% Entry condition and primary escalation rail.
\node[draw=hhebony,thick,fill=hhsand!55,rounded corners=2pt,

  {conflict reported};

\foreach \x/\n in {0/1,4.0/2,8.0/3,12.0/4} {
  \node[station] (s\n) at (\x,1.7) {\n};
}
\draw[->,>=Latex,very thick,draw=hhebony] (s1) -- (s2);
\draw[->,>=Latex,very thick,draw=hhebony] (s2) -- (s3);
\draw[->,>=Latex,very thick,draw=hhebony] (s3) -- (s4);

% Escalation triggers occupy a dedicated band above the rail.
\node[trigger,text width=31mm] at (2.0,2.54) {holder retains\\claim};
\node[trigger,text width=37mm] at (6.0,2.54) {waiting cost $>$ bond};
\node[trigger,text width=40mm] at (10.0,2.54) {evidence ambiguous\\or stakes $>$ ceiling};
\draw[hhamber,thick,dashed] (0.55,2.12) -- (3.45,2.12);

% Stage names below the rail.
\node[stagelabel] at (0,0.82) {Direct release};
\node[stagelabel] at (4.0,0.82) {TTL expiry};
\node[stagelabel] at (8.0,0.82) {Bonded\\preemption};
\node[stagelabel] at (12.0,0.82) {Human ruling};

\node[outcome] (o1) at (0,-0.35) {released by holder};

\node[outcome] (o3) at (8.0,-0.35) {preempt + settle};

\node[evid] at (6.0,-1.67)
  {evidence rail: resolution $\cdot$ actor $\cdot$ reason $\cdot$ artifact $\cdot$ outcome};

  {append to Merkle forest $\rightarrow$ open 24-hour appeal window};

\end{tikzpicture}%
}
\caption{Disputes follow one ordered escalation path. Each station is both a possible resolution and a gate to stronger intervention: direct release, TTL expiry, bonded preemption, then human ruling.
