% Reader question: when can Maya and the routing agent share one ranking?
% Claim: a strict reversal, not a difference in units or a tie, prevents
% one order from satisfying both readers at every selection budget.
% Evidence: synthetic ordinal ranks over three proposals named for real
% lib/ artifacts; the complete data is pdRankSamples below. Checked by
% tests/harbor-research/test_legible_swarm_split_ranker.py; no measured scores.
% Must distinguish: readers (left/right) from alternative cases (panel a / panel b);
% rank 1 is highest on BOTH sides. Only the briefing/attention pair reverses.
% Grammar: Tufte classical slopegraph, identical geometry and direct baseline labels.
% Reject: oversized geometric marker badges, raw fit-versus-loss units, or unified scores.
% Width: the 4.5-inch Book column, unscaled. All type inherits shared roles.
% Design references: https://charliepark.org/slopegraphs/ (ordinal scales,
% ties, direct labels); https://www.edwardtufte.com/notebook/
% sparkline-theory-and-practice-edward-tufte/ (words and graphics together).
% [internal, synthetic ordinal example]
\begin{figure}[H]
\centering
\pdfigurehue{pdteal}%
\begin{tikzpicture}[pd figure,x=1cm,y=1cm]
  \def\lxA{2.40}
  \def\rxA{4.20}
  \def\lxB{7.20}
  \def\rxB{9.00}
  \def\rankpitch{0.95}
  % artifact / routing rank / Maya agreement rank / Maya dispute rank / role tag
  \def\pdRankSamples{%
briefing.ts/1/1/2/briefing,
attention.ts/2/2/1/attention,
resurrection.ts/3/3/3/resurrection}

  % Top-level subtitle & dividing hairline
  \node[pd label,anchor=west,inner sep=0pt] at (0,4.40)
    {Synthetic ordinal ranks; 1 is highest on both sides};
  \draw[pd hairline] (0,4.10) -- (11.4,4.10);

  % =========================================================================
  % PANEL 1 (Left): Agreement (Concordant preferences across both observers)
  % =========================================================================
  \node[pd title,anchor=west,inner sep=0pt] at (0,3.75)
    {\textbf{(a) Agreement}};
  \node[pd label,anchor=west,inner sep=0pt,text=pdinkmuted] at (0,3.40)
    {one order serves both};

  % Column headers
  \node[pd label,anchor=south,align=center,inner sep=0pt] at (\lxA,2.65)
    {\textbf{Router}\\[-0.5ex]\textcolor{pdinkmuted}{\scriptsize fit}};
  \node[pd label,anchor=south,align=center,inner sep=0pt] at (\rxA,2.65)
    {\textbf{Maya}\\[-0.5ex]\textcolor{pdinkmuted}{\scriptsize regret}};

  % Guide spines
  \draw[draw=pdink!18,line width=0.4pt] (\lxA,2.50) -- (\lxA,0.30);
  \draw[draw=pdink!18,line width=0.4pt] (\rxA,2.50) -- (\rxA,0.30);

  % Agreement rows
  \foreach \artifact/\route/\agreement/\dispute/\role in \pdRankSamples {
    \pgfmathsetmacro{\yL}{2.40 - (\route-1)*\rankpitch}
    \pgfmathsetmacro{\yR}{2.40 - (\agreement-1)*\rankpitch}
    \draw[draw=pdink!55,line width=0.75pt] (\lxA,\yL) -- (\rxA,\yR);
    \fill[pdink!70] (\lxA,\yL) circle (1.2pt);
    \fill[pdink!70] (\rxA,\yR) circle (1.2pt);
    \node[pd label,anchor=east,align=right,inner sep=0pt]
      at ({\lxA-0.12},\yL)
      {\texttt{\artifact}\enspace\textbf{\route}};
    \node[pd label,anchor=west,align=left,inner sep=0pt]
      at ({\rxA+0.12},\yR)
      {\textbf{\agreement}};
  }

  % =========================================================================
  % PANEL 2 (Right): Dispute (Strict reversal preventing single-ranking)
  % =========================================================================
  \node[pd title,anchor=west,inner sep=0pt] at (6.8,3.75)
    {\textbf{(b) Dispute}};
  \node[pd label,anchor=west,inner sep=0pt,text=pdinkmuted] at (6.8,3.40)
    {top choices conflict};

  % Column headers
  \node[pd label,anchor=south,align=center,inner sep=0pt] at (\lxB,2.65)
    {\textbf{Router}\\[-0.5ex]\textcolor{pdinkmuted}{\scriptsize fit}};
  \node[pd label,anchor=south,align=center,inner sep=0pt] at (\rxB,2.65)
    {\textbf{Maya}\\[-0.5ex]\textcolor{pdinkmuted}{\scriptsize regret}};

  % Guide spines
  \draw[draw=pdink!18,line width=0.4pt] (\lxB,2.50) -- (\lxB,0.30);
  \draw[draw=pdink!18,line width=0.4pt] (\rxB,2.50) -- (\rxB,0.30);

  % Dispute rows
  \foreach \artifact/\route/\agreement/\dispute/\role in \pdRankSamples {
    \pgfmathsetmacro{\yL}{2.40 - (\route-1)*\rankpitch}
    \pgfmathsetmacro{\yR}{2.40 - (\dispute-1)*\rankpitch}
    \ifnum\route=\dispute\relax
      % Unchanged item (resurrection.ts): subtle dashed baseline
      \draw[draw=pdink!35,line width=0.6pt,dash pattern=on 2pt off 1.5pt]
        (\lxB,\yL) -- (\rxB,\yR);
      \fill[pdink!40] (\lxB,\yL) circle (1.1pt);
      \fill[pdink!40] (\rxB,\yR) circle (1.1pt);
      \node[pd label,anchor=east,align=right,inner sep=0pt,text=pdinkmuted]
        at ({\lxB-0.12},\yL)
        {\textbf{\route}};
      \node[pd label,anchor=west,align=left,inner sep=0pt,text=pdinkmuted]
        at ({\rxB+0.12},\yR)
        {\textbf{\dispute}\enspace\texttt{\artifact}};
    \else
      % Reversing items: bold highlighted crossing slopes
      \ifnum\route=1\relax
        % briefing.ts drops 1 -> 2 (Router top choice, Maya rank 2)
        \draw[draw=pdcobalt,line width=1.2pt] (\lxB,\yL) -- (\rxB,\yR);
        \fill[pdcobalt] (\lxB,\yL) circle (1.4pt);
        \fill[pdcobalt] (\rxB,\yR) circle (1.4pt);
        \node[pd label,anchor=east,align=right,inner sep=0pt,text=pdcobalt]
          at ({\lxB-0.12},\yL)
          {\textbf{\route}};
        \node[pd label,anchor=west,align=left,inner sep=0pt,text=pdcobalt]
          at ({\rxB+0.12},\yR)
          {\textbf{\dispute}\enspace\texttt{\artifact}};
      \else
        % attention.ts rises 2 -> 1 (Maya top choice, Router rank 2)
        \draw[draw=pdteal,line width=1.2pt] (\lxB,\yL) -- (\rxB,\yR);
        \fill[pdteal] (\lxB,\yL) circle (1.4pt);
        \fill[pdteal] (\rxB,\yR) circle (1.4pt);
        \node[pd label,anchor=east,align=right,inner sep=0pt,text=pdteal]
          at ({\lxB-0.12},\yL)
          {\textbf{\route}};
        \node[pd label,anchor=west,align=left,inner sep=0pt,text=pdteal]
          at ({\rxB+0.12},\yR)
          {\textbf{\dispute}\enspace\texttt{\artifact}};
      \fi
    \fi
  }
\end{tikzpicture}
\caption{One pair crosses when Maya and the router disagree. Panel (a) shows concordant preferences (one ranking satisfies both readers); panel (b) shows the strict reversal where high irreversible risk inverts the top choices. Panels compare alternative objective preferences, not time.}
\label{fig:split-ranker}
\end{figure}
