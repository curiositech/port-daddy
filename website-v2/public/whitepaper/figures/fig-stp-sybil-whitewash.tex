% Figure III.6 — free labels branch; bound identities return to one principal.
% Visual question: where do sanctions land after a respawn or a Sybil split?
\begin{figure}[H]
\centering
\resizebox{0.94\textwidth}{!}{%
\begin{tikzpicture}[font=\small,node distance=14mm]
  \node[stp heading] (freeh) {SELF-ASSERTED LABELS};
  \node[stp box,text width=30mm,below=6mm of freeh] (bad) {bad outcome\\{\scriptsize label X}};
  \node[stp box,text width=27mm,right=of bad,yshift=8mm] (x1) {fresh X$'$\\{\scriptsize clean record}};
  \node[stp box,text width=27mm,right=of bad,yshift=-8mm] (xmany) {$X_1\ldots X_n$\\{\scriptsize forged quorum}};
  \draw[stp dashed arrow] (bad) -- (x1);
  \draw[stp dashed arrow] (bad) -- (xmany);
  \node[stp label,below=9mm of bad] {sanction discarded with the label};

  \node[stp heading,right=42mm of freeh] (boundh) {BOUND PRINCIPAL ROOT};
  \node[stp accent box,text width=34mm,below=6mm of boundh] (principal)
    {principal record\\{\scriptsize owns bonds and history}};
  \node[stp box,text width=26mm,right=of principal,yshift=8mm] (i1) {identity A\\{\scriptsize process 1}};
  \node[stp box,text width=26mm,right=of principal,yshift=-8mm] (i2) {identity B\\{\scriptsize respawn}};
  \draw[stp accent arrow] (principal) -- (i1);
  \draw[stp accent arrow] (principal) -- (i2);
  \draw[stp dashed arrow] (i2.south) -- ++(0,-9mm) -| (principal.south);
  \node[stp label,below=9mm of i2] {sanction returns to the same root};
\end{tikzpicture}}
\caption{\textbf{Identity binding determines whether sanctions survive churn.}
Self-asserted labels branch freely into a clean replacement or a forged quorum.
With a daemon-minted identity bound to one economic principal, every descendant
process returns outcomes and sanctions to the same durable root; respawning does
not reset the ledger.}
\label{fig:sybil}
\end{figure}
