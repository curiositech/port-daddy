% Brief (tikz-diagram-craft). Register row ch7-14 (must).
% Reader question: what evidence lets a verifier walk from one ordered note to
%   one signed harbor epoch, and how large is that evidence?
% Claim: the receipt contains two distinct inclusion proofs: seven sibling
%   hashes to the 100-note session root, then fourteen sibling hashes to the
%   10,000-session harbor root, plus one 64-byte signature -- 736 bytes total.
% Grammar chosen: a four-step vertical proof journey. Each row makes one claim;
%   the byte arithmetic is a separate final row. This deliberately spends page
%   height to preserve readable labels and visible arrow shafts.
% Five-second test: read 7 hashes, then 14 hashes, then 64-byte signature, then
%   point to the 736-byte receipt.
\pdfigurehue{pdgold}%
\begin{figure*}[t]
\centering
\begin{tikzpicture}[pd figure,x=1cm,y=1cm]
  \tikzset{
    proof node/.style={pd state,minimum width=2.45cm,minimum height=1.02cm,
      inner xsep=8pt,inner ysep=7pt,align=center},
    proof packet/.style={pd artifact,minimum width=3.05cm,minimum height=1.02cm,
      inner xsep=8pt,inner ysep=7pt,align=center},
    proof root/.style={pd focus outline,minimum width=2.60cm,minimum height=1.08cm,
      inner xsep=8pt,inner ysep=7pt,align=center},
    proof signed/.style={pd focus state,double=pdpage,double distance=1.1pt,
      minimum width=2.80cm,minimum height=1.12cm,inner xsep=8pt,
      inner ysep=7pt,align=center},
    proof step/.style={pd kind,anchor=west,text width=2.55cm,align=left},
    proof why/.style={pd note,anchor=west,text width=2.65cm,align=left},
    proof arrow/.style={pd focus arrow,shorten <=1pt,shorten >=2pt},
    sum term/.style={pd state,minimum width=3.05cm,minimum height=1.12cm,
      inner xsep=7pt,inner ysep=7pt,align=center},
    sum result/.style={pd climax state,minimum width=2.20cm,minimum height=1.12cm,
      inner xsep=7pt,inner ysep=7pt,align=center}
  }

  \node[pd title,anchor=west] at (0.00,7.80)
    {one note becomes one signed, portable receipt};

  % 1. The chain commits to order; it is not an inclusion proof.
  \node[proof step] at (0.00,6.58) {1\quad ORDER};
  \node[proof node] (prior) at (4.20,6.25) {prior hash};
  \node[pd climax state,minimum width=2.45cm,minimum height=1.02cm,
    inner xsep=8pt,inner ysep=7pt] (note) at (7.20,6.25) {note};
  \node[proof node] (nhash) at (10.20,6.25) {note hash};
  \node[proof why] at (12.45,6.25) {Fixes the note's position in the session log.};

  % 2. First inclusion path: note to session root.
  \node[proof step] at (0.00,4.78) {2\quad SESSION PROOF};
  \node[proof node] (noteleaf) at (4.20,4.45) {note hash\\{\ttfamily leaf}};
  \node[proof packet] (seven) at (7.55,4.45)
    {7 sibling hashes\\for 100 notes};
  \node[proof root] (session) at (11.05,4.45) {session root};
  \node[proof why] at (12.45,4.45) {One of 100 ordered notes.};

  % 3. Second inclusion path: session root to signed harbor root.
  \node[proof step] at (0.00,2.98) {3\quad HARBOR PROOF};
  \node[proof root] (sessionleaf) at (4.20,2.65) {session root\\{\ttfamily leaf}};
  \node[proof packet] (fourteen) at (7.55,2.65)
    {14 sibling hashes\\for 10,000 sessions};
  \node[proof signed] (harbor) at (11.05,2.65) {signed\\harbor root};
  \node[proof why] at (12.45,2.65) {One of 10,000 sessions. Signed by the harbor.};

  % Long, straight shafts stay visible at final Book size. Connectors are
  % painted before the filled nodes so no arrow crosses text.
  \begin{scope}[on background layer]
    \draw[proof arrow] (prior.east) -- (note.west);
    \draw[proof arrow] (note.east) -- (nhash.west);
    \draw[proof arrow] (noteleaf.east) -- (seven.west);
    \draw[proof arrow] (seven.east) -- (session.west);
    \draw[proof arrow] (sessionleaf.east) -- (fourteen.west);
    \draw[proof arrow] (fourteen.east) -- (harbor.west);
  \end{scope}

  % 4. The receipt budget is a separate comparison, not squeezed into the
  % proof-path rail above.
  \draw[pd rule] (0.00,1.62) -- (15.10,1.62);
  \node[proof step] at (0.00,0.92) {4\quad RECEIPT SIZE};
  \node[sum term] at (4.20,0.58)
    {7 hashes $\times$ 32 B\\\textbf{224 B}};
  \node[pd title] at (5.95,0.58) {+};
  \node[sum term] at (7.70,0.58)
    {14 hashes $\times$ 32 B\\\textbf{448 B}};
  \node[pd title] at (9.45,0.58) {+};
  \node[sum term] at (11.20,0.58)
    {Ed25519 signature\\\textbf{64 B}};
  \node[sum result] at (14.00,0.58)
    {\textbf{736 B}\\total};
