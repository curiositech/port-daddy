\begin{tikzpicture}[
  font=\small,
  ent/.style={draw=hhebony,thick,fill=hhsand!30,rounded corners=2pt,inner sep=4pt,
              align=center,minimum width=26mm,minimum height=11mm},
  ledger/.style={draw=hhink,thick,fill=hhsand!30,rounded corners=2pt,inner sep=4pt,
                 align=center,font=\footnotesize,minimum height=11mm},

% top row: work -> select -> judge -> bond
\node[ent] (work) at (0,0) {graded\\\textbf{work}};
\node[ent] (select) at (3.6,0) {select judge:\\\emph{conflict-free}\\\emph{influence-free}};
\node[ent,fill=hhsand!30,draw=hhink] (judge) at (7.4,0) {\textbf{neutral judge}\\{\scriptsize ``university /}\\{\scriptsize rating agency''}};
\node[ent] (bond) at (11.2,0) {judge posts\\\textbf{own bond}\\{\scriptsize + reputation}};
\draw[arr] (work) -- (select);
\draw[arr] (select) -- (judge);
\draw[arr] (judge) -- (bond);

% bottom row: grade -> outcome ledger ; re-audit slashes the bond
\node[ledger] (out) at (7.4,-2.6) {grade enters the\\\textbf{outcome ledger}};
\node[ent,fill=hhsand!30,draw=hhcobalt] (reaudit) at (11.2,-2.6) {sampled\\\textbf{re-audit}};
\draw[arr,draw=hhink] (judge) -- (out);
\draw[arr,draw=hhink] (out) -- (reaudit);
\draw[slash] (reaudit) -- node[midway,right=1mm,font=\scriptsize,text=hhink,align=left]
  {overturn\\$\Rightarrow$ \textbf{slash}} (bond);
\end{tikzpicture}}
\caption{\textbf{The neutral-judge market.} Graded work flows to a judge selected
for conflict- and influence-freedom; the judge posts its own bond, the grade enters
the outcome ledger, and a sampled re-audit can slash a judge whose grade is
overturned. ``Neutral'' is a checklist (conflict-free, influence-free, de-biased),
not a vibe, and the judge is hired through the same bonded ceremony the market uses
for any task (Protocol~\ref{proto:judge}) --- the judge's own reputation is the
asset it protects.}
\label{fig:judge-market}
