% Figure III.11 — three explicit selection gates before a grade is admitted.
% Visual question: what makes a judge eligible and accountable?
\begin{figure}[H]
\centering
\resizebox{0.94\textwidth}{!}{%
\begin{tikzpicture}[font=\small,node distance=13mm]
  \node[stp box,text width=28mm] (work) {ungraded\\work};
  \node[stp box,text width=32mm,right=of work] (conflict)
    {\textbf{conflict screen}\\{\scriptsize no shared principal or stake}};
  \node[stp box,text width=32mm,right=of conflict] (blind)
    {\textbf{blind review}\\{\scriptsize author metadata withheld}};
  \node[stp accent box,text width=32mm,right=of blind] (bond)
    {\textbf{bonded judge}\\{\scriptsize signs a vector grade}};
  \node[stp box,text width=28mm,right=of bond] (ledger) {outcome\\ledger};
  \draw[stp arrow] (work) -- (conflict);
  \draw[stp arrow] (conflict) -- (blind);
  \draw[stp accent arrow] (blind) -- (bond);
  \draw[stp arrow] (bond) -- (ledger);
  \node[stp box,text width=32mm,below=15mm of ledger] (audit)
    {sampled re-audit\\{\scriptsize overturn $\Rightarrow$ slash}};
  \draw[stp arrow] (ledger) -- (audit);
  \draw[stp dashed arrow] (audit.west) -| (bond.south);
  \node[stp label,above=7mm of conflict] {eligibility};
  \node[stp label,above=7mm of blind] {influence control};
  \node[stp label,above=7mm of bond] {accountability};
\end{tikzpicture}}
\caption{\textbf{Neutrality is a gated ceremony.} Work reaches a judge only
after conflict screening and blind presentation; the selected judge posts a bond
and signs the grade before it enters the outcome ledger. Sampled re-audit closes
the loop by making an overturned grade costly to the judge itself.}
\label{fig:judge-market}
\end{figure}
