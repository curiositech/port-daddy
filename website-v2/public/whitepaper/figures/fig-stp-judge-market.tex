  \draw[stp arrow] (judge) -- (bond);

  \node[stp label] at (4.3,-0.75) {overturn $\Rightarrow$ slash};
\end{tikzpicture}}
\caption{\textbf{The neutral-judge market.} Graded work flows to a judge selected
for conflict- and influence-freedom; the judge posts its own bond, the grade enters
the outcome ledger, and a sampled re-audit can slash a judge whose grade is
overturned. ``Neutral'' is a checklist (conflict-free, influence-free, de-biased),
not a vibe, and the judge is hired through the same bonded ceremony the market uses
for any task (Protocol~\ref{proto:judge}) --- the judge's own reputation is the
asset it protects.}
\label{fig:judge-market}
