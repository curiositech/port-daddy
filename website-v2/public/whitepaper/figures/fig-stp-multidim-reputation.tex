% Figure III.10 — a source-owned three-axis quality vector.
% Visual question: why must the score remain a vector rather than one scalar?
\begin{figure}[H]
\centering
\resizebox{0.90\textwidth}{!}{%
\begin{tikzpicture}[font=\small]
  \coordinate (c) at (0,0);
  \coordinate (acc) at (0,3.0);
  \coordinate (aes) at (-2.9,-1.8);
  \coordinate (eff) at (2.9,-1.8);
  \draw[hhgray,line width=.55pt] (acc) -- (aes) -- (eff) -- cycle;
  \draw[hhgray!55,line width=.45pt] ($(c)!.5!(acc)$) -- ($(c)!.5!(aes)$) -- ($(c)!.5!(eff)$) -- cycle;
  \draw[hhink,line width=.65pt] (c) -- (acc);
  \draw[hhink,line width=.65pt] (c) -- (aes);
  \draw[hhink,line width=.65pt] (c) -- (eff);
  \fill[hhcobalt] (c) circle (2.4pt);
  \node[stp accent box,text width=33mm] at (c) {$\mathbf q=(q_{\rm acc},q_{\rm aes},q_{\rm eff})$};
  \node[stp box,text width=35mm,anchor=south] at ([yshift=5mm]acc)
    {\textbf{accuracy}\\{\scriptsize acceptance-test oracle}};
  \node[stp box,text width=35mm,anchor=east] at ([xshift=-6mm]aes)
    {\textbf{aesthetics}\\{\scriptsize independent reviewer}};
  \node[stp box,text width=35mm,anchor=west] at ([xshift=6mm]eff)
    {\textbf{efficiency}\\{\scriptsize cost ledger}};
  \node[stp label,anchor=north] at (0,-3.2)
    {buyer applies a preference vector $\mathbf w$ only after disclosure};
\end{tikzpicture}}
\caption{\textbf{Quality remains a three-axis disclosure.} Accuracy,
aesthetics, and efficiency have different competent judges and cannot be collapsed
without hiding trade-offs. The ledger publishes $\mathbf q$ intact; each buyer
then applies its own preference vector $\mathbf w$ rather than inheriting a single
Goodhart-able global score.}
\label{fig:multidim}
\end{figure}
