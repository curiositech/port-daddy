% Fig 10 — preserve the quality vector; let each buyer choose its weights.
\begin{figure}[H]
\centering
\resizebox{0.94\textwidth}{!}{%
\begin{tikzpicture}[font=\small]
  \node[stp box,minimum width=38mm] (out) at (-5,1.8)
    {\textbf{witnessed outcome}};
  \node[stp accent box,minimum width=40mm] (vector) at (0,1.8)
    {\textbf{quality vector}\\{\small $\mathbf q=(q_{\rm acc},q_{\rm aes},q_{\rm eff})$}};
  \node[stp box,minimum width=42mm,text width=36mm] (buyer) at (5.2,1.8)
    {\textbf{buyer applies $\mathbf w$}\\
     {\scriptsize ``cheap'' versus ``careful''}};
  \draw[stp accent arrow] (out) -- (vector);
  \draw[stp accent arrow] (vector) -- (buyer);

  \node[stp box,minimum width=43mm,text width=36mm] (acc) at (-5,-1.25)
    {\textbf{accuracy} $q_{\rm acc}$\\{\scriptsize judged by a test oracle}};
  \node[stp box,minimum width=43mm,text width=36mm] (aes) at (0,-1.25)
    {\textbf{aesthetics} $q_{\rm aes}$\\{\scriptsize expert-reviewer judgment}};
  \node[stp box,minimum width=43mm,text width=36mm] (eff) at (5,-1.25)
    {\textbf{efficiency} $q_{\rm eff}$\\{\scriptsize judged by the cost ledger}};
  \draw[stp arrow] (vector.south) -- ++(0,-5mm) -| (acc.north);
  \draw[stp arrow] (vector.south) -- (aes.north);
  \draw[stp arrow] (vector.south) -- ++(0,-5mm) -| (eff.north);
\end{tikzpicture}}
\caption{\textbf{Multi-dimensional reputation.} A witnessed outcome
carries a quality \emph{vector} $\mathbf{q}=(q_{\text{acc}},q_{\text{aes}},
q_{\text{eff}})$. Each axis is scored by a \emph{different} competent judge: a test
oracle for accuracy, a senior reviewer for aesthetics, the cost meter for
efficiency. The buyer reads the vector and applies their own weights --- which is
exactly what avoids collapsing quality into one Goodhart-able scalar.}
\label{fig:multidim}
\end{figure}
