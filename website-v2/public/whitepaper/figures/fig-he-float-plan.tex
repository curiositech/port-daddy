  \node[pd row label,text width=1.8cm,align=right] at (-.4,1.6) {escrow row};
  \node[pd row label,text width=1.8cm,align=right] at (-.4,.2) {worker state};

  % The commit: one focus rule crossing all three rows.
  \draw[pd focus rule,line width=1.6pt] (4.5,-.65)--(4.5,3.9);

  \draw[pd rule,line width=1.2pt] (-.3,3.0)--(4.5,3.0)--(4.5,3.5);

  \node[pd direct label,inner sep=2.6pt,anchor=south] at (1.0,3.08) {funded};
  \node[pd direct label,inner sep=2.6pt,anchor=south] at (6.0,3.58) {debited};

  \draw[pd rule,dashed,line width=1.0pt] (-.3,.2)--(4.5,.2)--(4.5,.7);

\caption{\textbf{Execution is admissible only at one committed transition.} The
three rows are the wallet, the escrow row, and the worker's process state over
Protocol~\ref{prot:escrow}'s float-plan ceremony. Before the commit they read
funded / no row / no process; the wallet debit and escrow-row creation happen at
the same commit, and only after it does the worker move to \texttt{running}. A
signed plan alone, or an attempted debit that never commits, is insufficient: the
dashed abort path returns to the start with all three traces unchanged
[internal].}
\label{fig:he-float-plan}
