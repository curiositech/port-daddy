% Swiss edition overrides: grotesk labels, flat colour blocks in the chapter's
% hue, no outlines, heavy marks, right angles. Loaded by pd-figure-language.tex.
\tikzset{
  pd figure/.append style={font=\pdgrotesk\footnotesize,text=hhink},
  pd hairline/.append style={draw=hhink!45,line width=.5pt},
  pd rule/.append style={draw=hhink,line width=.9pt},
  pd focus rule/.append style={draw=\pdcurrentchaptercolor,line width=2pt},
  pd caution rule/.append style={draw=pderror,line width=2pt},
  pd guide/.append style={draw=hhink!40,line width=.5pt},
  pd arrow/.append style={-{Triangle[length=2.4mm,width=2.2mm]},draw=hhink,line width=1.1pt},
  pd focus arrow/.append style={-{Triangle[length=2.4mm,width=2.2mm]},draw=\pdcurrentchaptercolor,line width=1.5pt},
  pd caution arrow/.append style={-{Triangle[length=2.4mm,width=2.2mm]},draw=pderror,line width=1.5pt},
  pd row label/.append style={font=\pdgrotesk\footnotesize\bfseries,text=hhink,anchor=east},
  pd panel title/.append style={font=\pdgrotesk\footnotesize\bfseries,text=hhink,align=left},
  pd axis label/.append style={font=\pdgrotesk\footnotesize,text=hhink},
  pd direct label/.append style={font=\pdgrotesk\footnotesize,text=hhink,align=left,inner sep=1.5pt,fill=hhpaper},
  pd note/.append style={font=\pdgrotesk\footnotesize,text=hhink!70!hhgray,align=left},
  pd tick/.append style={draw=hhink,line width=.6pt},
  pd datum/.append style={rectangle,fill=hhink,draw=none,inner sep=2.3pt},
  pd focus datum/.append style={rectangle,fill=\pdcurrentchaptercolor,draw=none,inner sep=2.7pt},
  pd caution datum/.append style={rectangle,fill=pderror,draw=none,inner sep=2.7pt},
  pd state/.append style={draw=none,fill=\pdcurrentchaptercolor,text=hhpaper,rounded corners=0pt,
    font=\pdgrotesk\footnotesize\bfseries,inner xsep=6pt,inner ysep=4pt,minimum height=6.5mm},
  pd terminal/.append style={pd state,fill=hhink},
  pd actor/.append style={draw=hhink,line width=.9pt,fill=hhpaper,rounded corners=0pt,inner xsep=6pt,inner ysep=4pt},
  pd artifact/.append style={draw=none,fill=hhsand,rounded corners=0pt,inner xsep=6pt,inner ysep=4pt},
  pd boundary/.append style={draw=hhink,line width=.9pt,rounded corners=0pt},
  pd focus fill/.append style={fill=\pdcurrentchaptercolor!22,draw=none},
  pd caution fill/.append style={fill=pderror!20,draw=none},
  pd neutral fill/.append style={fill=hhsand,draw=none},
  % A hatch stays a hatch.  This line used to read `fill=hhink!12,draw=none`,
  % which is not a Swiss restyling of a texture but its deletion: in TikZ a
  % `fill=` set after the base style's `pattern=` cancels the pattern outright,
  % so `pd hatch` rendered here as a flat 12 per cent wash.  Measured on a
  % 4x1cm patch at 1.0x/150dpi, the luminance standard deviation across the
  % patch was 0.00 under swiss against 18.11 with no edition loaded and 50.66
  % under technical -- a perfectly uniform field, which is not a texture
  % flattened but a texture removed.  Swiss keeps a pattern, in Swiss
  % character: a heavier rule at the same 45 degrees, and no outline.
  pd hatch/.append style={pattern={Lines[angle=45,distance=4.2pt,line width=.6pt]},
    pattern color=hhink!58,draw=none},
}
