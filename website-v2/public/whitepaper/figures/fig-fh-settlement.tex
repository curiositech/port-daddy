\begin{tikzpicture}[
  party/.style={draw=hhink, line width=0.65pt, fill=hhsand!25, rounded corners=2pt, inner sep=6pt, minimum width=36mm, minimum height=18mm, align=center},
  escrow/.style={draw=hhcobalt, line width=0.85pt, fill=hhcobalt!7, rounded corners=2pt, inner sep=6pt, minimum width=48mm, minimum height=18mm, align=center},
  stepbox/.style={draw=hhink, line width=0.5pt, fill=hhpaper, rounded corners=2pt, inner sep=5pt, font=\scriptsize, text width=42mm, align=left},
  outcomebox/.style={draw=hhink, line width=0.6pt, fill=hhsand!35, rounded corners=2pt, inner sep=5pt, font=\scriptsize, text width=36mm, align=center},
  guardbox/.style={draw=hhcobalt!70, line width=0.65pt, fill=hhcobalt!5, rounded corners=2pt, inner sep=5pt, font=\scriptsize, text width=42mm, align=left},
  flowarr/.style={-{Stealth[length=2.0mm,width=1.3mm]}, draw=hhink, line width=0.55pt},
  accentarr/.style={-{Stealth[length=2.0mm,width=1.3mm]}, draw=hhcobalt, line width=0.75pt}
]

  % ─── Parties at Top ─────────────────────────────────────────────────────────
  \node[party] (PA) at (-4.8, 0) {
    \textbf{\scshape Alice (Principal)}\\[1pt]
    {\scriptsize Performance Bond Poster}
  };
  \node[escrow] (E) at (0.0, 0) {
    \textbf{\scshape Settlement Escrow}\\[1pt]
    {\scriptsize Custody bounded \emph{only if}}\\
    {\scriptsize Prop.~\ref{thm:fh-escrow-bound}'s assumptions hold}
  };
  \node[party] (PB) at (4.8, 0) {
    \textbf{\scshape Bob (Principal)}\\[1pt]
    {\scriptsize Bounty \& Damage Claimant}
  };

  % ─── Steps 1, 2, 3 Below ────────────────────────────────────────────────────
  \node[stepbox] (s1) at (-4.8, -2.4) {
    \textbf{1. Bond Posting}\\[1pt]
    $A$ mints work card for agent $\alpha$; posts performance bond $B_\alpha$ into the escrow.
  };

  \node[stepbox] (s2) at (0.0, -2.4) {
    \textbf{2. Damage Claim}\\[1pt]
    Breach detected at $B$; Bob signs claim $\mathsf{cl}_B = (R_B^{(e)}, \text{evidence})$ to witness log.
  };

  \node[stepbox] (s3) at (4.8, -2.4) {
    \textbf{3. Oracle Verification}\\[1pt]
    Escrow checks inclusion of $\alpha$'s actions in $R_B^{(e)}$ against step 1 criteria.
  };

  \draw[flowarr] (PA.south) -- (s1.north);
  \draw[flowarr] (E.south) -- (s2.north);
  \draw[flowarr] (PB.south) -- (s3.north);

  \draw[accentarr] (s1.east) -- (s2.west);
  \draw[accentarr] (s2.east) -- (s3.west);

  % ─── Outcomes and Structural Guard ──────────────────────────────────────────
  \node[guardbox] (guard) at (-4.8, -4.8) {
    \textbf{\scshape Custody Assumption:}\\
    \emph{If} whitelist, fee cap, and single-terminal-transition are non-bypassable (Prop.~\ref{thm:fh-escrow-bound}), funds cannot be redirected. Not established here for a deployed system.
  };

  \node[outcomebox] (Op) at (0.5, -4.8) {
    \textbf{\scshape Settlement: Clear}\\[1pt]
    $B$ compensated from bond; remainder refunded to $A$.
  };

  \node[outcomebox] (Or) at (4.8, -4.8) {
    \textbf{\scshape Settlement: Refuse}\\[1pt]
    Bond returned to $A$; escrow logs evidence proof.
  };

  \draw[flowarr] (s3.south) -- ++(0, -0.6) -| (Op.north);
  \draw[flowarr] (s3.south) -- ++(0, -0.6) -| (Or.north);

\end{tikzpicture}
\caption{Cross-harbor settlement protocol. The custody bounds are \emph{conditional}, not established: Property~\ref{thm:fh-escrow-bound} holds only where the recipient whitelist, fee cap, and single-terminal-transition rule cannot be bypassed by the escrow operator (\S\ref{sec:fh-escrow}); 2-of-3 threshold signing is one candidate way to arrange that. Under those assumptions the outcome space partitions into \textbf{Clear} or \textbf{Refuse}, bounding cross-harbor liability; without them the worst case is the full custodial balance.}
\label{fig:fh-settlement}
\end{figure}
