% "No Partial Reanimation". The claim of this figure is a MISSING edge, so the
% absence is drawn explicitly (crossed-out arrow) rather than merely omitted:
% a reader must not be able to mistake it for an oversight.
%
% No trust boundary here by design — this is the verifier's own state, not a
% message crossing a channel. See fig-anchor-alg-confusion.tex for the
% convention and the figures that need it.

\begin{figure}[H]
\centering
\begin{tikzpicture}[
  font=\small,
  state/.style={draw=hhebony,thick,fill=hhsand,rounded corners=3pt,
                minimum width=26mm,minimum height=11mm,align=center,inner sep=4pt},
  terminal/.style={state,fill=hhsanddeep},
  edge/.style={->,>=Latex,thick,draw=hhink},
  never/.style={->,>=Latex,thick,draw=hhcobalt,dashed},
  edgelbl/.style={font=\footnotesize,align=center},
  note/.style={draw=hhebony,thick,fill=hhpaper,rounded corners=3pt,
               inner sep=5pt,align=left,font=\footnotesize,text width=140mm},
]

\node[state] (valid) at (0,0) {\textbf{valid}\\[1pt]{\footnotesize admitted}};
\node[terminal] (revoked) at (5.6,1.5) {\textbf{revoked}\\[1pt]{\footnotesize rejected}};
\node[terminal] (expired) at (5.6,-1.5) {\textbf{expired}\\[1pt]{\footnotesize rejected}};

\draw[edge] (valid.north east) -- node[edgelbl,above left=-1mm and -4mm]
  {operator or daemon\\revokes \texttt{kid}} (revoked.west);
\draw[edge] (valid.south east) -- node[edgelbl,below left=-1mm and -4mm]
  {clock passes\\$t_0 + \tau$} (expired.west);
\draw[edge] (revoked.south) -- node[edgelbl,right=1mm]
  {$t_0+\tau$; entry\\may leave $R$} (expired.north);

% The claim: there is NO way back. Drawn, not omitted.
\draw[never] (revoked.north west) to[bend right=38]
  node[pos=0.5,fill=hhpaper,draw=hhcobalt,thick,circle,inner sep=1pt,
       font=\bfseries,text=hhcobalt] (crossmark) {$\times$}
  (valid.north);
\node[edgelbl,text=hhcobalt,anchor=south,text width=52mm]
  at ([yshift=3mm]crossmark.north)
  {\textbf{never} --- no un-revocation. Property~\ref{prop:no-reanim}: an
   erroneous revocation is repaired by \emph{issuing a new card}, not by
   restoring this one.};

\node[note,below=12mm of valid.south,anchor=north west,xshift=-13mm]
  {{\bfseries Why the missing edge matters.}\quad
   Acceptance is monotonically narrowing over a card's TTL
   (Property~\ref{prop:filter-mono}), so a verifier that has once rejected a
   \texttt{kid} inside $[t_0, t_0{+}\tau]$ will never later admit it. This is
   what lets the revocation check be a pure \emph{gate} in front of admission,
   and why the soundness argument for
   Theorem~\ref{thm:injective} survives revocation unchanged.};

\end{tikzpicture}
\caption{The Harbor Card lifecycle. A card leaves the \textit{valid} state in
exactly two ways --- an explicit revocation, or the clock reaching $t_0+\tau$ ---
and returns from neither. The crossed dashed arrow is the figure's actual claim:
the $\textit{revoked} \to \textit{valid}$ transition is \emph{absent by design}
(Property~\ref{prop:no-reanim}), not merely undrawn. After expiry the entry may
be dropped from the revocation log $R$ without weakening anything, because the
card is already unacceptable on its \texttt{exp} claim alone.}
\label{fig:anchor-card-lifecycle}
\end{figure}
