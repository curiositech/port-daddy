  % ─── Phase 1: Symmetric Pinning ─────────────────────────────────────────────
  \node[phasecard] (p1) at (-6.3, 0) {
    \textbf{\scshape Phase 1: HS256}\\[1pt]
    {\scriptsize\textsc{Symmetric Pinning}}\\[3pt]
    {\scriptsize Alg header ignored; key pinned at issuance. Header trust = 0.}
  };
  \node[threatcard, below=4mm of p1] (t1) {
    \textbf{Forecloses:}\\
    Alg confusion attacks (CVE-2015-9235, CVE-2023-48238).
  };

  % ─── Phase 2: Asymmetric Identity ───────────────────────────────────────────
  \node[phasecard] (p2) at (-2.1, 0) {
    \textbf{\scshape Phase 2: Ed25519}\\[1pt]
    {\scriptsize\textsc{Asymmetric Identity}}\\[3pt]
    {\scriptsize Daemon holds Root CA; private key never leaves issuer.}
  };
  \node[threatcard, below=4mm of p2] (t2) {
    \textbf{Forecloses:}\\
    Shared-secret theft and cross-harbor key compromise.
  };

  % ─── Phase 3: Stigmergic Delegation ─────────────────────────────────────────
  \node[phasecard] (p3) at (2.1, 0) {
    \textbf{\scshape Phase 3: Macaroon}\\[1pt]
    {\scriptsize\textsc{Stigmergic Delegation}}\\[3pt]
    {\scriptsize Multi-hop attenuation ($\text{cap}_k \subseteq \text{cap}_{k-1}$) with nonce-chain binding.}
  };
  \node[threatcard, below=4mm of p3] (t3) {
    \textbf{Forecloses:}\\
    Capability escalation, hop splicing, and chain substitution.
  };

  % ─── Phase 4: Dynamic Revocation ────────────────────────────────────────────
  \node[phaseaccent] (p4) at (6.3, 0) {
    \textbf{\scshape Phase 4: Cuckoo}\\[1pt]
    {\scriptsize\textsc{Dynamic Revocation}}\\[3pt]
    {\scriptsize Append-only revocation log + Cuckoo read-cache. Epoch-gossiped.}
  };
  \node[threatcard, below=4mm of p4] (t4) {
    \textbf{Forecloses:}\\
    Post-issuance compromise and stale authorization windows.
  };

  \draw[steparr] (p1.east) -- (p2.west);
