% Semantic contract: every phase closes one additional threat while retaining
% every earlier closure. The staircase is the mechanism; color is redundant.
\begin{figure}[H]
\centering
\begin{tikzpicture}[pd figure,x=1cm,y=1cm]
  \def\xmin{-4.55}
  \def\xmax{5.25}
  \foreach \y/\threat/\close in {
    2.1/{header-selected\\algorithm}/-3.25,
    .7/{shared-secret theft}/-1.05,
    -.7/{downstream\\privilege growth}/1.15,
    -2.1/{post-issue authority}/3.35} {
    \node[pd axis label,anchor=east,align=right,text width=31mm] at (\xmin-0.18,\y) {\threat};
    \draw[pd hairline] (\xmin,\y) -- (\xmax,\y);
    \draw[draw=hhamber,line width=2.4pt,line cap=round] (\xmin,\y) -- (\close,\y);
    \draw[draw=hhpaper,line width=1.0pt,line cap=round] (\xmin,\y) -- (\close,\y);
    \draw[draw=hhteal,line width=2.4pt,line cap=round] (\close,\y) -- (\xmax,\y);
    \node[pd caution datum,minimum size=4.8pt,inner sep=0pt] at (\close,\y) {};
  }

  \draw[pd caution rule]
    (-3.25,2.1) -- (-3.25,.7) -- (-1.05,.7) -- (-1.05,-.7) --
    (1.15,-.7) -- (1.15,-2.1) -- (3.35,-2.1);
  \node[pd direct label,text=hhamber,anchor=south west] at (-3.15,2.28)
    {closure frontier};

  \foreach \x/\n/\name in {
    -3.25/1/{HS256 pinning},
    -1.05/2/{Ed25519 identity},
    1.15/3/{delegation},
    3.35/4/{revocation}} {
    \draw[pd tick] (\x,-2.48) -- (\x,-2.65);
    \node[pd axis label,anchor=north,align=center,text width=28mm] at (\x,-2.72)
      {\textbf{Phase \n}\\\name};
  }
  \draw[pd rule,-{Stealth[length=1.7mm,width=1.05mm]}]
    (\xmin,-2.55) -- (\xmax,-2.55);
  \node[pd note,anchor=west] at (-4.5,-3.55)
    {amber before closure; teal thereafter; no protection reopens};
\end{tikzpicture}
\caption{Threat closure is cumulative, not substitutive. Each protocol phase
moves one attack surface across the closure frontier while every earlier rail
stays closed. The staircase therefore encodes the architectural invariant: a
later mechanism adds protection without reopening a hole already closed.}
\label{fig:anchor-four-phases}
\end{figure}
