% Redraw of fig-sealed-two-worlds to the tikz-diagram-craft v2 standard.
%
% Reader question: do two runs that differ only in Derek's secret look
% identical to Erin, the observer?
% Claim: yes at every step but the declared release, and even there, for two
% secrets of equal parity, the release still agrees.
% Evidence: Theorem sealed-noninterference and its proof idea, and "Numbers by
% hand" (both s=0 and s=2 give g(s)=0), website-v2/public/whitepaper/
% sealed-harbor.tex, Sec.~sealed-noninterference [internal, c1_noninterference.py].
% Must distinguish: the four lockstep steps where Erin's view is tied (=)
% from the one step (release, boxed as "the slot") the theorem actually lets
% differ; the honest run (ties hold) from the mutation (a leaky gate, where
% they would not).
% Grammar chosen: two-run lockstep, before/after on identical geometry
% (tpl-before-after) -- one time axis, two lanes, so the same step compares
% directly across worlds.
% Grammar rejected: a single trace with a branch would hide that the compared
% quantity is Erin's view of two SEPARATE runs, not one run's two options.
% Five-second test: which step is the only one allowed to differ, and does it.
\begin{figure}[H]
\centering
\pdfigurehue{pdcobalt}%
\begin{tikzpicture}[pd figure,x=1cm,y=1cm]
  \def\lx{0}      % load column
  \def\rx{2.1}    % read column: leave a visible arrow shaft between boxes
  \def\sx{4.2}    % submit column
  \def\gx{6.4}    % release column, inside the slot
  \def\yA{6.6}    % World A lane
  \def\yB{3.4}    % World B lane
  % --- the two worlds, identical except Derek's secret ---------------------
  \node[pd title,anchor=east,align=center,text width=1.5cm] at (-1.0,\yA) {World A\\$s=0$};
  \node[pd title,anchor=east,align=center,text width=1.5cm] at (-1.0,\yB) {World B\\$s=2$};
  \node[pd state,minimum width=1.42cm] (A1) at (\lx,\yA) {load};
  \node[pd state,minimum width=1.42cm] (A2) at (\rx,\yA) {read};
  \node[pd state,minimum width=1.42cm] (A3) at (\sx,\yA) {submit};
  \node[pd state,minimum width=1.42cm] (B1) at (\lx,\yB) {load};
  \node[pd state,minimum width=1.42cm] (B2) at (\rx,\yB) {read};
  \node[pd state,minimum width=1.42cm] (B3) at (\sx,\yB) {submit};
  \draw[pd focus arrow] (A1) -- (A2); \draw[pd focus arrow] (A2) -- (A3);
  \draw[pd focus arrow] (B1) -- (B2); \draw[pd focus arrow] (B2) -- (B3);
  % --- Erin's observable state, tied at every step ($\equiv$), drawn as two
  % short segments so the mark they frame never sits on a stroked path -----
  \foreach \x in {\lx,\rx,\sx} {
    \draw[pd hairline] (\x,{\yA-.36}) -- (\x,{\yA-.90});
    \draw[pd hairline] (\x,{\yB+.90}) -- (\x,{\yB+.36});
    \node[pd label] at (\x,{(\yA+\yB)/2}) {$\equiv$};
  }
  % --- the slot: the one declared release step ------------------------------
  \node[pd focus state,line width=1.6pt,minimum width=1.42cm,align=center] (A4) at (\gx,\yA) {release\\$g=0$};
  \node[pd focus state,line width=1.6pt,minimum width=1.42cm,align=center] (B4) at (\gx,\yB) {release\\$g=0$};
  \begin{scope}[on background layer]
    \node[pd panel,fit={(A4)(B4)},inner xsep=2pt] (slot) {};
  \end{scope}
  \node[pd title,anchor=south] at (slot.north) {the slot};
  \draw[pd focus arrow] (A3) -- (A4); \draw[pd focus arrow] (B3) -- (B4);
  \draw[pd hairline] (\gx,{\yA-.60}) -- (\gx,{\yA-.90});
  \draw[pd hairline] (\gx,{\yB+.90}) -- (\gx,{\yB+.60});
  \node[pd label] at (\gx,{(\yA+\yB)/2}) {$\equiv$};
  \node[pd note,anchor=north west,align=left,text width=3.3cm] at (-0.50,1.95)
    {Erin's trace is identical at every step, to depth 7};
  % --- the mutation: honest gate replaced by one that leaks the raw secret -
  \node[pd breach state,text width=2.1cm,align=center] (LEAK) at (\gx,-0.05)
    {leaky gate:\\caught here};
  \draw[pd breach arrow] (slot.south) -- (LEAK.north);
\end{tikzpicture}
\caption{Two runs of the clean room, identical except Derek's secret ($s{=}0$ against $s{=}2$, equal parity so $g(s){=}0$ either way), driven in lockstep through \texttt{load}$\to$\texttt{read}$\to$\texttt{submit}$\to$\texttt{release}. Erin's observable state is tied ($\equiv$) at every step; the release step is boxed as ``the slot'' because it is the only step the theorem lets differ at all, and here it still does not. Below the slot, replacing the honest gate with one that leaks the raw secret breaks the tie at this same step -- leak $0$ from World A against leak $2$ from World B, now distinguishable. The finite clean-room model, secrets $s\in\{0,1,2,3\}$ with $g(s)=s\bmod 2$, passes this check exhaustively to depth 7 over every interleaving and secret pair; both mutations are caught -- the leaked-secret gate shown here (4 steps) and a worker that bypasses the gate entirely (3 steps) [internal, \texttt{c1\_noninterference.py}].}
\label{fig:sealed-two-worlds}
\end{figure}
