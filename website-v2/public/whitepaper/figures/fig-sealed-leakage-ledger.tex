% from contracted channels to total adversarial capacity.
%
% Reader question: what capacity did the contract grant across 200 jobs, and
% exactly what do three scheduled return times add?
% Claim: the status channel grants 12,800 bits before timing; three scheduled
% return slots add about 317 bits; the engagement therefore grants about
% 13,117 bits of adversarial capacity in total.
% Evidence: Sec. sealed-budget's worked example and Exercise sealed-budget:
% 200 x 64 = 12,800; 200 x log2(3) = 316.99; total = 13,116.99 bits.
% Must distinguish: the declared status capacity from the separately priced
% timing capacity, while making their addition auditable without a compressed
% scientific-notation axis.
% Grammar chosen: a two-row accounting flow into one ledger bucket. Each row
% carries its own units and arithmetic, then both feed the visible total.
% Grammar rejected: the former waterfall, whose 317-bit increment was only
% 2.5 percent of the 12,800-bit base and therefore nearly invisible at Book
% size even though it was the figure's point.
% Five-second test: read 12,800 + 317 = 13,117 directly from the drawing.
\begin{figure}[H]
\centering
\pdfigurehue{pdcobalt}%
\begin{tikzpicture}[pd figure,x=1cm,y=1cm]
  \def\yA{1.92}
  \def\yB{0.35}
  \node[pd title,anchor=south west] at (0.05,3.18)
    {Capacity ledger for one contract};
  \node[pd kind] at (1.55,2.75) {1\quad contract terms};
  \node[pd kind] at (4.72,2.75) {2\quad channel capacity};
  \node[pd kind] at (8.85,2.75) {3\quad add to ledger};

  \node[pd focus state,text width=2.65cm] (STATUS) at (1.55,\yA)
    {$200$ jobs $\times$ $64$ bits per status};
  \node[pd focus outline,text width=2.50cm] (SBITS) at (4.72,\yA)
    {status channel\\\textbf{12{,}800 bits}};
  \draw[pd focus arrow] (STATUS) -- (SBITS);

  \node[pd warn state,text width=2.65cm] (TIMING) at (1.55,\yB)
    {$200$ returns $\times$ $1.585$ bits per return};
  \node[pd warn state,text width=2.50cm] (TBITS) at (4.72,\yB)
    {timing channel\\\textbf{$+317$ bits}};
  \draw[pd warn arrow] (TIMING) -- (TBITS);

  \node[pd focus badge] (SUM) at (7.05,1.14) {$+$};
  \draw[pd focus arrow,pd to badge] (SBITS.east) -- (6.35,\yA) |- (SUM.north);
  \draw[pd warn arrow,pd to badge] (TBITS.east) -- (6.35,\yB) |- (SUM.south);
  \node[pd bucket,minimum width=3.05cm,minimum height=2.15cm]
    (TOTAL) at (9.50,1.14) {\\[6pt]total account\\\textbf{13{,}117 bits}};
  \draw[pd focus arrow,pd from badge] (SUM.east) -- (TOTAL.west);
  \node[pd title,anchor=north] at (8.55,-0.43)
    {$12{,}800 + 317 = 13{,}117$ bits};
