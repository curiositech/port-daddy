% fig-ch0-progressive-skills.tex -- redrawn to the house figure language (v2).
% Reader question: which layer of a skill is loaded how often, and what
% does each layer cost in context?
% Claim: a skill's four layers -- frontmatter, the SKILL.md body,
% references, and scripts/schemas -- are read at activation frequencies
% that contract by roughly a decade at each step (P=1.0, 10^-1, 10^-3,
% 10^-4); the deepest two layers are never loaded into the language
% model's context at all -- the agent queries references by byte offset
% and runs scripts in a host subshell, so their own large token volume is
% not the number that matters.
% Grammar chosen: a step chart (tpl-step-chart.tex's grammar) on a
% logarithmic frequency axis, never interpolated between the four
% categorical layers; each step's own context-token cost is a direct
% label, a second quantity riding the same step rather than a second axis.
% Rejected: the "iceberg" of four coloured cards with a browser/gear/key
% icon, a flowchart-document icon, an open-book icon, a terminal, and a
% chest-mounted robot with a rainbow token-meter gauge and a "MASSIVE
% CONTEXT SAVINGS" banner -- none of those icons distinguish a decade of
% activation frequency from another, and the gauge needle has no stated
% units; a log-scale staircase says "each layer is read ten times less
% often than the one above it" in one line a reader can trace with a
% finger, which no icon panel can.
% Facts: website-v2/public/whitepaper/coordination-papers-mega-volume-chapter-0.tex
% Sec. "Context Economics and Progressive Skill Disclosure" -- Tier 1
% (frontmatter, P=1.0, cached prefix), Tier 2 (SKILL.md body, P~=10^-1),
% Tier 3 (references/ and scripts/, P~=10^-3 to 10^-4, "the agent does
% not load Tier 3 assets into its language context") -- and the figure's
% existing \pdwidecaption.
\pdfigurehue{pdcobalt}%
\begin{tikzpicture}[pd figure]
\begin{axis}[name=freqplot,pd axis,
  width=10.0cm,height=5.6cm,
  ymode=log,log basis y={10},
  xmin=-0.25,xmax=3.55,ymin=0.00002,ymax=6,
  xtick={0,1,2,3},
  xticklabels={frontmatter,{SKILL.md body},references,{scripts/schemas}},
  x tick label style={font=\pdfiglabel,align=center,yshift=-2pt},
  ytick={1,0.1,0.001,0.0001},
  yticklabels={$10^{0}$,$10^{-1}$,$10^{-3}$,$10^{-4}$},
  ylabel={activation frequency (decades)}]
  \addplot[pd series,const plot,mark=none]
    coordinates {(0,1)(1,0.1)(2,0.001)(3,0.0001)(3.4,0.0001)};
  \node[pd label,align=center,anchor=south,yshift=9pt] at (axis cs:0.5,1)
    {$\sim$100 tok/skill\\(cached prefix)};
  \node[pd label,align=center,anchor=south,yshift=9pt] at (axis cs:1.5,0.1)
    {$\sim$1k--10k tokens\\(while active)};
  \node[pd label,align=left,anchor=south west] at (axis cs:2.15,0.006)
    {references \& scripts/schemas:\\0 tokens loaded into context};
  \node[pd focus datum] at (axis cs:3.2,0.0001) {};
  \node[pd focus label,anchor=south west,yshift=8pt] at (axis cs:3.15,0.0001)
    {0 tokens};
\end{axis}
\node[pd note,anchor=north west,text width=13.0cm] at ([yshift=-0.55cm]freqplot.outer south west)
  {references and scripts are Tier 3: the agent queries them by offset or
   runs them in a host subshell, so their own large token volume --
   $10^4$--$10^5$ tokens apiece -- is never the number that reaches the
   language model.};
\end{tikzpicture}
