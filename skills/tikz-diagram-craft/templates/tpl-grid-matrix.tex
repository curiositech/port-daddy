\begin{tikzpicture}[pd figure,x=1cm,y=1cm]
  \def\c{1.18}   % cell pitch

  \def\x0{2.85}
  \foreach \r/\name/\hits in {0/{peer A}/{1,4}, 1/{peer B}/{2,5}, 2/{union}/{1,2,4,5}}{

\caption{Two peers insert the same id \texttt{c1} into a six-bucket filter and
land it in different buckets, an artefact of each peer's own insertion and
eviction history. The bottom row is what merging must produce: the union of the
placements. OR-ing the two bit arrays would instead claim four independent
memberships, which is why daemons gossip the id set rather than the array.
[internal]}
